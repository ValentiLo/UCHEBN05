# Автоматический стартер для проекта с авторизацией и БД

Вот несколько способов автоматически создать проект:

## **Способ 1: Bash-скрипт для автоматического создания**

### **1. Создайте файл `create_project.sh`:**

```bash
#!/bin/bash

# Автоматический создатель проекта с авторизацией
# Использование: ./create_project.sh имя_проекта

# Проверка аргументов
if [ -z "$1" ]; then
    echo "Использование: $0 имя_проекта"
    exit 1
fi

PROJECT_NAME=$1
PROJECT_DIR="$PROJECT_NAME"

# Цвета для вывода
RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
NC='\033[0m' # No Color

echo -e "${YELLOW}Создание проекта: $PROJECT_NAME...${NC}"

# Создание структуры папок
mkdir -p "$PROJECT_DIR"
mkdir -p "$PROJECT_DIR/includes"
mkdir -p "$PROJECT_DIR/admin"
mkdir -p "$PROJECT_DIR/assets/css"
mkdir -p "$PROJECT_DIR/assets/js"
mkdir -p "$PROJECT_DIR/assets/images/profiles"
mkdir -p "$PROJECT_DIR/logs"

# Файл подключения к БД
cat > "$PROJECT_DIR/includes/database.php" << 'EOF'
<?php
require_once 'config.php';

class Database {
    private static $instance = null;
    private $connection;
    
    private function __construct() {
        try {
            $dsn = "mysql:host=" . DB_HOST . ";dbname=" . DB_NAME . ";charset=utf8mb4";
            $this->connection = new PDO($dsn, DB_USER, DB_PASS);
            $this->connection->setAttribute(PDO::ATTR_ERRMODE, PDO::ERRMODE_EXCEPTION);
            $this->connection->setAttribute(PDO::ATTR_DEFAULT_FETCH_MODE, PDO::FETCH_ASSOC);
            $this->connection->setAttribute(PDO::ATTR_EMULATE_PREPARES, false);
        } catch (PDOException $e) {
            die("Ошибка подключения: " . $e->getMessage());
        }
    }
    
    public static function getInstance() {
        if (self::$instance === null) {
            self::$instance = new Database();
        }
        return self::$instance;
    }
    
    public function getConnection() {
        return $this->connection;
    }
    
    public function execute($sql, $params = []) {
        $stmt = $this->connection->prepare($sql);
        $stmt->execute($params);
        return $stmt;
    }
}

function db() {
    return Database::getInstance();
}
?>
EOF

<?php
require_once 'database.php';

class Auth {
    private $db;
    
    public function __construct() {
        $this->db = Database::getInstance()->getConnection();
    }
    
    public function register($data) {
        $data['password_hash'] = password_hash($data['password'], PASSWORD_DEFAULT);
        unset($data['password'], $data['password_confirm']);
        
        $sql = "INSERT INTO users (username, email, password_hash, first_name, last_name) 
                VALUES (:username, :email, :password_hash, :first_name, :last_name)";
        
        $this->db->prepare($sql)->execute($data);
        $userId = $this->db->lastInsertId();
        
        $user = $this->getUserById($userId);
        $this->createSession($user);
        
        return ['success' => true, 'user' => $user];
    }
    
    public function login($username, $password) {
        $user = $this->getUserByUsernameOrEmail($username);
        
        if (!$user || !password_verify($password, $user['password_hash'])) {
            return ['success' => false, 'message' => 'Неверные данные'];
        }
        
        $this->createSession($user);
        $this->updateLastLogin($user['id']);
        
        return ['success' => true, 'user' => $user];
    }
    
    public function logout() {
        session_unset();
        session_destroy();
        session_start();
        $_SESSION['csrf_token'] = bin2hex(random_bytes(32));
    }
    
    public function isLoggedIn() {
        return isset($_SESSION['user_id']) && isset($_SESSION['user_role']);
    }
    
    public function getCurrentUser() {
        if (!$this->isLoggedIn()) return null;
        
        $sql = "SELECT * FROM users WHERE id = ?";
        $stmt = $this->db->prepare($sql);
        $stmt->execute([$_SESSION['user_id']]);
        return $stmt->fetch();
    }
    
    public function hasRole($role) {
        $user = $this->getCurrentUser();
        return $user && $user['role'] === $role;
    }
    
    private function getUserByUsernameOrEmail($identifier) {
        $sql = "SELECT * FROM users WHERE username = ? OR email = ?";
        $stmt = $this->db->prepare($sql);
        $stmt->execute([$identifier, $identifier]);
        return $stmt->fetch();
    }
    
    private function getUserById($id) {
        $sql = "SELECT * FROM users WHERE id = ?";
        $stmt = $this->db->prepare($sql);
        $stmt->execute([$id]);
        return $stmt->fetch();
    }
    
    private function createSession($user) {
        $_SESSION['user_id'] = $user['id'];
        $_SESSION['user_role'] = $user['role'];
    }
    
    private function updateLastLogin($userId) {
        $sql = "UPDATE users SET last_login = NOW() WHERE id = ?";
        $this->db->prepare($sql)->execute([$userId]);
    }
}

$auth = new Auth();
?>
EOF

    $db_name = $_POST['db_name'];
    $db_user = $_POST['db_user'];
    $db_pass = $_POST['db_pass'];
    $site_name = $_POST['site_name'];
    $admin_email = $_POST['admin_email'];
    $admin_password = $_POST['admin_password'];
    
    // Подключение к БД
    try {
        $dsn = "mysql:host=$db_host;charset=utf8mb4";
        $pdo = new PDO($dsn, $db_user, $db_pass);
        $pdo->setAttribute(PDO::ATTR_ERRMODE, PDO::ERRMODE_EXCEPTION);
        
        // Создание БД
        $pdo->exec("CREATE DATABASE IF NOT EXISTS `$db_name`");
        $pdo->exec("USE `$db_name`");
        
        // Создание таблиц
        $sql = file_get_contents('database.sql');
        $pdo->exec($sql);

                "define('DB_PASS', '');",
                "define('SITE_NAME', 'Мой сайт');"

                "define('DB_NAME', '$db_name');",
                "define('DB_USER', '$db_user');",

        echo '<div class="alert alert-success">Установка завершена успешно!</div>';
```

## **Автоматическая установка с проверками:**

Вот улучшенный скрипт, который сам все проверит:

```bash
#!/bin/bash
# auto_installer.sh

# Проверка зависимостей
check_dependencies() {
    echo "Проверка зависимостей..."
    
    # PHP
    if ! command -v php &> /dev/null; then
        echo "❌ PHP не установлен"
        exit 1
    fi
    
    # MySQL
    if ! command -v mysql &> /dev/null; then
        echo "❌ MySQL не установлен"
        exit 1
    fi
    
    # Composer (опционально)
    if command -v composer &> /dev/null; then
        echo "✓ Composer установлен"
    fi
    
    echo "✓ Все зависимости установлены"
}

# Автоматическая установка
main() {
    PROJECT_NAME=${1:-"my_project"}
    
    check_dependencies
    
    echo "Создание проекта: $PROJECT_NAME"
    
    # Создание структуры
    mkdir -p "$PROJECT_NAME"
    
    # Копирование готовых файлов
    # (Здесь должна быть логика копирования или генерации файлов)
    
    echo "✓ Проект создан"
    echo ""
    echo "Для запуска:"
    echo "  cd $PROJECT_NAME"
    echo "  php -S localhost:8000"
}

main "$@"
```
