## **1. Файл database.sql (структура БД)**

```sql
-- Создание базы данных
CREATE DATABASE IF NOT EXISTS my_site_db;
USE my_site_db;

-- Таблица пользователей
CREATE TABLE users (
    id INT PRIMARY KEY AUTO_INCREMENT,
    username VARCHAR(50) UNIQUE NOT NULL,
    email VARCHAR(100) UNIQUE NOT NULL,
    password_hash VARCHAR(255) NOT NULL,
    first_name VARCHAR(50),
    last_name VARCHAR(50),
    role ENUM('admin', 'user', 'moderator') DEFAULT 'user',
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    last_login TIMESTAMP NULL,
    profile_image VARCHAR(255) DEFAULT 'default.jpg'
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- Таблица сессий
CREATE TABLE user_sessions (
    id INT PRIMARY KEY AUTO_INCREMENT,
    user_id INT NOT NULL,
    session_token VARCHAR(255) UNIQUE NOT NULL,
    ip_address VARCHAR(45),
    user_agent TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    expires_at TIMESTAMP NOT NULL,
    FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- Таблица неудачных попыток входа
CREATE TABLE failed_login_attempts (
    id INT PRIMARY KEY AUTO_INCREMENT,
    ip_address VARCHAR(45) NOT NULL,
    attempt_time TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    username VARCHAR(50),
    INDEX idx_ip_time (ip_address, attempt_time)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- Таблица для сброса пароля
CREATE TABLE password_resets (
    id INT PRIMARY KEY AUTO_INCREMENT,
    user_id INT NOT NULL,
    token VARCHAR(255) NOT NULL,
    expires_at TIMESTAMP NOT NULL,
    used BOOLEAN DEFAULT FALSE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE,
    INDEX idx_token (token)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
```

## **3. Файл includes/database.php**

```php
<?php
require_once 'config.php';

class Database {
    private static $instance = null;
    private $connection;
    
    private function __construct() {
        try {
            $dsn = "mysql:host=" . DB_HOST . ";dbname=" . DB_NAME . ";charset=utf8mb4";
            $this->connection = new PDO($dsn, DB_USER, DB_PASS);
            $this->connection->setAttribute(PDO::ATTR_ERRMODE, PDO::ERRMODE_EXCEPTION);
            $this->connection->setAttribute(PDO::ATTR_DEFAULT_FETCH_MODE, PDO::FETCH_ASSOC);
            $this->connection->setAttribute(PDO::ATTR_EMULATE_PREPARES, false);
        } catch (PDOException $e) {
            die("Ошибка подключения к базе данных: " . $e->getMessage());
        }
    }
    
    public static function getInstance() {
        if (self::$instance === null) {
            self::$instance = new Database();
        }
        return self::$instance;
    }
    
    public function getConnection() {
        return $this->connection;
    }
    
    public function prepare($sql) {
        return $this->connection->prepare($sql);
    }
    
    // Безопасное выполнение запроса
    public function execute($sql, $params = []) {
        try {
            $stmt = $this->prepare($sql);
            $stmt->execute($params);
            return $stmt;
        } catch (PDOException $e) {
            $this->logError($e->getMessage(), $sql);
            throw $e;
        }
    }
    
    // Получение одной записи
    public function fetch($sql, $params = []) {
        $stmt = $this->execute($sql, $params);
        return $stmt->fetch();
    }
    
    // Получение всех записей
    public function fetchAll($sql, $params = []) {
        $stmt = $this->execute($sql, $params);
        return $stmt->fetchAll();
    }
    
    // Вставка записи
    public function insert($table, $data) {
        $columns = implode(', ', array_keys($data));
        $placeholders = ':' . implode(', :', array_keys($data));
        $sql = "INSERT INTO $table ($columns) VALUES ($placeholders)";
        
        $this->execute($sql, $data);
        return $this->connection->lastInsertId();
    }
    
    // Обновление записи
    public function update($table, $data, $where, $whereParams = []) {
        $set = [];
        foreach ($data as $key => $value) {
            $set[] = "$key = :$key";
        }
        $setClause = implode(', ', $set);
        $sql = "UPDATE $table SET $setClause WHERE $where";
        
        $params = array_merge($data, $whereParams);
        $this->execute($sql, $params);
    }
    
    // Логирование ошибок
    private function logError($message, $sql = '') {
        $logMessage = date('Y-m-d H:i:s') . " - $message";
        if ($sql) {
            $logMessage .= " - SQL: $sql";
        }
        file_put_contents(__DIR__ . '/../logs/db_errors.log', $logMessage . PHP_EOL, FILE_APPEND);
    }
}

// Глобальная функция для удобства
function db() {
    return Database::getInstance();
}
?>
```

## **4. Файл includes/auth.php**

```php
<?php
require_once 'database.php';

class Auth {
    private $db;
    
    public function __construct() {
        $this->db = Database::getInstance()->getConnection();
    }
    
    // Регистрация нового пользователя
    public function register($data) {
        // Валидация данных
        $errors = $this->validateRegistration($data);
        if (!empty($errors)) {
            return ['success' => false, 'errors' => $errors];
        }
        
        // Проверка уникальности
        if ($this->userExists($data['username'], $data['email'])) {
            return ['success' => false, 'errors' => ['username' => 'Пользователь с таким именем или email уже существует']];
        }
        
        // Хеширование пароля
        $data['password_hash'] = password_hash($data['password'], PASSWORD_DEFAULT);
        unset($data['password'], $data['password_confirm']);
        
        // Создание пользователя
        try {
            $userId = db()->insert('users', $data);
            
            // Автоматический вход после регистрации
            $user = $this->getUserById($userId);
            $this->createSession($user);
            
            return [
                'success' => true,
                'message' => 'Регистрация успешна!',
                'user' => $user
            ];
        } catch (PDOException $e) {
            return ['success' => false, 'errors' => ['system' => 'Ошибка при регистрации']];
        }
    }
    
    // Вход пользователя
    public function login($username, $password) {
        // Проверка блокировки по IP
        if ($this->isIpBlocked()) {
            return ['success' => false, 'message' => 'Слишком много попыток входа. Попробуйте позже.'];
        }
        
        // Поиск пользователя
        $user = $this->getUserByUsernameOrEmail($username);
        
        if (!$user || !$user['is_active']) {
            $this->logFailedAttempt($_SERVER['REMOTE_ADDR'], $username);
            return ['success' => false, 'message' => 'Неверное имя пользователя или пароль'];
        }
        
        // Проверка пароля
        if (!password_verify($password, $user['password_hash'])) {
            $this->logFailedAttempt($_SERVER['REMOTE_ADDR'], $username);
            return ['success' => false, 'message' => 'Неверное имя пользователя или пароль'];
        }
        
        // Создание сессии
        $this->createSession($user);
        
        // Обновление времени последнего входа
        db()->update('users', ['last_login' => date('Y-m-d H:i:s')], 'id = :id', ['id' => $user['id']]);
        
        // Очистка неудачных попыток
        $this->clearFailedAttempts($_SERVER['REMOTE_ADDR']);
        
        return ['success' => true, 'message' => 'Вход выполнен успешно', 'user' => $user];
    }
    
    // Выход пользователя
    public function logout() {
        if (isset($_SESSION['user_id']) && isset($_SESSION['session_token'])) {
            // Удаление сессии из БД
            $sql = "DELETE FROM user_sessions WHERE user_id = ? AND session_token = ?";
            db()->execute($sql, [$_SESSION['user_id'], $_SESSION['session_token']]);
        }
        
        // Очистка сессии
        session_unset();
        session_destroy();
        
        // Создание новой сессии (для CSRF токена)
        session_start();
        $_SESSION['csrf_token'] = bin2hex(random_bytes(32));
        
        return true;
    }
    
    // Проверка авторизации
    public function isLoggedIn() {
        if (!isset($_SESSION['user_id']) || !isset($_SESSION['session_token'])) {
            return false;
        }
        
        // Проверка сессии в БД
        $sql = "SELECT * FROM user_sessions 
                WHERE user_id = ? AND session_token = ? AND expires_at > NOW()";
        $session = db()->fetch($sql, [$_SESSION['user_id'], $_SESSION['session_token']]);
        
        return !empty($session);
    }
    
    // Получение текущего пользователя
    public function getCurrentUser() {
        if (!$this->isLoggedIn()) {
            return null;
        }
        
        $sql = "SELECT id, username, email, first_name, last_name, role, profile_image 
                FROM users WHERE id = ? AND is_active = TRUE";
        return db()->fetch($sql, [$_SESSION['user_id']]);
    }
    
    // Проверка роли
    public function hasRole($role) {
        $user = $this->getCurrentUser();
        return $user && $user['role'] === $role;
    }
    
    // Проверка нескольких ролей
    public function hasAnyRole($roles) {
        $user = $this->getCurrentUser();
        return $user && in_array($user['role'], $roles);
    }
    
    // Вспомогательные методы
    private function validateRegistration($data) {
        $errors = [];
        
        if (strlen($data['username']) < 3 || strlen($data['username']) > 50) {
            $errors['username'] = 'Имя пользователя должно быть от 3 до 50 символов';
        }
        
        if (!filter_var($data['email'], FILTER_VALIDATE_EMAIL)) {
            $errors['email'] = 'Неверный формат email';
        }
        
        if (strlen($data['password']) < 6) {
            $errors['password'] = 'Пароль должен быть не менее 6 символов';
        }
        
        if ($data['password'] !== $data['password_confirm']) {
            $errors['password_confirm'] = 'Пароли не совпадают';
        }
        
        return $errors;
    }
    
    private function userExists($username, $email) {
        $sql = "SELECT COUNT(*) as count FROM users WHERE username = ? OR email = ?";
        $result = db()->fetch($sql, [$username, $email]);
        return $result['count'] > 0;
    }
    
    private function getUserByUsernameOrEmail($identifier) {
        $sql = "SELECT * FROM users WHERE (username = ? OR email = ?) AND is_active = TRUE";
        return db()->fetch($sql, [$identifier, $identifier]);
    }
    
    private function getUserById($id) {
        $sql = "SELECT id, username, email, first_name, last_name, role, profile_image 
                FROM users WHERE id = ? AND is_active = TRUE";
        return db()->fetch($sql, [$id]);
    }
    
    private function createSession($user) {
        // Генерация токена сессии
        $sessionToken = bin2hex(random_bytes(32));
        $expiresAt = date('Y-m-d H:i:s', time() + SESSION_DURATION);
        
        // Сохранение в БД
        $sessionData = [
            'user_id' => $user['id'],
            'session_token' => $sessionToken,
            'ip_address' => $_SERVER['REMOTE_ADDR'],
            'user_agent' => $_SERVER['HTTP_USER_AGENT'] ?? '',
            'expires_at' => $expiresAt
        ];
        db()->insert('user_sessions', $sessionData);
        
        // Установка сессионных переменных
        $_SESSION['user_id'] = $user['id'];
        $_SESSION['session_token'] = $sessionToken;
        $_SESSION['user_role'] = $user['role'];
    }
    
    private function isIpBlocked() {
        $ip = $_SERVER['REMOTE_ADDR'];
        $timeThreshold = date('Y-m-d H:i:s', time() - LOCKOUT_TIME);
        
        $sql = "SELECT COUNT(*) as attempts 
                FROM failed_login_attempts 
                WHERE ip_address = ? AND attempt_time > ?";
        $result = db()->fetch($sql, [$ip, $timeThreshold]);
        
        return $result['attempts'] >= MAX_LOGIN_ATTEMPTS;
    }
    
    private function logFailedAttempt($ip, $username = null) {
        db()->insert('failed_login_attempts', [
            'ip_address' => $ip,
            'username' => $username
        ]);
    }
    
    private function clearFailedAttempts($ip) {
        $sql = "DELETE FROM failed_login_attempts WHERE ip_address = ?";
        db()->execute($sql, [$ip]);
    }
}

// Глобальный экземпляр авторизации
$auth = new Auth();
?>
```

```php
            const input = index === 0 ? passwordInput : passwordConfirmInput;
```

## **Инструкция по установке:**

### **1. Установка базы данных:**
```bash
mysql -u root -p < database.sql
```

### **2. Настройка конфигурации:**
Отредактируйте файл `includes/config.php`:
- Укажите правильные данные для подключения к БД
- Настройте `SITE_URL` и `SITE_NAME`
- При необходимости измените настройки безопасности

### **3. Структура проекта:**
Создайте необходимые папки:
```bash
mkdir -p assets/{css,js,images/profiles}
mkdir -p admin
mkdir -p logs
chmod 755 logs
```

### **4. Генерация хешей паролей:**
Для тестовых пользователей в БД сгенерируйте хеши:
```php
// В файле database.sql замените:
// '$2y$10$YourHashedPasswordHere'
// На реальный хеш, например:
$hash = password_hash('admin123', PASSWORD_DEFAULT);
// Результат: $2y$10$92IXUNpkjO0rOQ5byMi.Ye4oKoEa3Ro9llC/.og/at2.uheWG/igi
```

### **5. Запуск:**
1. Поместите проект в папку веб-сервера (например, `htdocs`)
2. Откройте в браузере: `http://localhost/my-site/`
3. Используйте тестовые учетные записи:
   - Админ: `admin / admin123`
   - Пользователь: `user / user123`

### **6. Особенности безопасности:**
- Все пароли хешируются с помощью `password_hash()`
- Защита от SQL-инъекций через подготовленные выражения
- CSRF-токены для всех форм
- Ограничение попыток входа
- HTTPS рекомендуется на продакшене

Этот шаблон предоставляет полнофункциональную систему авторизации, которую можно легко расширять и модифицировать под конкретные нужды проекта.
