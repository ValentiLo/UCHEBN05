### **Ключевые понятия:**

| Термин | Определение | Практическое значение |
|--------|------------|----------------------|
| **Аутентификация** | Проверка подлинности пользователя (кто вы?) | Логин/пароль, биометрия, 2FA |
| **Авторизация** | Проверка прав доступа (что вам можно?) | Роли, разрешения, ACL |
| **Сессия** | Временное состояние аутентифицированного пользователя | Cookie, JWT, сессионные токены |

### **Модели авторизации:**

1. **RBAC (Role-Based Access Control)** — на основе ролей
2. **ABAC (Attribute-Based Access Control)** — на основе атрибутов
3. **MAC (Mandatory Access Control)** — обязательный контроль
4. **DAC (Discretionary Access Control)** — дискреционный контроль

### **Уровни доступа в типичном веб-приложении:**

CREATE TABLE users (
    id INT PRIMARY KEY AUTO_INCREMENT,
    username VARCHAR(50) UNIQUE NOT NULL,
    email VARCHAR(100) UNIQUE NOT NULL,
    password_hash VARCHAR(255) NOT NULL,
    is_active BOOLEAN DEFAULT TRUE,

    id INT PRIMARY KEY AUTO_INCREMENT,
    name VARCHAR(50) UNIQUE NOT NULL,
    description TEXT

    id INT PRIMARY KEY AUTO_INCREMENT,

    session_token VARCHAR(255) UNIQUE NOT NULL,

### **Задание 2: Реализация регистрации и входа**

**Файл: `auth.php`**

```php
<?php
session_start();
require_once 'db.php';

class Auth {
    private $db;
    
    public function __construct($db) {
        $this->db = $db;
    }
    
    // Регистрация пользователя
    public function register($username, $email, $password) {
        // Валидация данных
        if (strlen($password) < 8) {
            return ['success' => false, 'message' => 'Пароль должен быть не менее 8 символов'];
        }
        
        // Хеширование пароля
        $passwordHash = password_hash($password, PASSWORD_DEFAULT);
        
        // Проверка существования пользователя
        $stmt = $this->db->prepare("SELECT id FROM users WHERE username = ? OR email = ?");
        $stmt->execute([$username, $email]);
        
        if ($stmt->rowCount() > 0) {
            return ['success' => false, 'message' => 'Пользователь с таким именем или email уже существует'];
        }
        
        // Создание пользователя
        $stmt = $this->db->prepare("INSERT INTO users (username, email, password_hash) VALUES (?, ?, ?)");
        $stmt->execute([$username, $email, $passwordHash]);
        
        // Назначение роли "user" по умолчанию
        $userId = $this->db->lastInsertId();
        $roleId = $this->getRoleId('user');
        
        $stmt = $this->db->prepare("INSERT INTO user_roles (user_id, role_id) VALUES (?, ?)");
        $stmt->execute([$userId, $roleId]);
        
        return ['success' => true, 'message' => 'Регистрация успешна'];
    }
    
    // Вход пользователя
    public function login($username, $password) {
        // Получение пользователя
        $stmt = $this->db->prepare("SELECT * FROM users WHERE username = ? AND is_active = TRUE");
        $stmt->execute([$username]);
        $user = $stmt->fetch(PDO::FETCH_ASSOC);
        
        if (!$user) {
            return ['success' => false, 'message' => 'Неверное имя пользователя или пароль'];
        }
        
        // Проверка пароля
        if (!password_verify($password, $user['password_hash'])) {
            // Логирование неудачной попытки
            $this->logFailedAttempt($user['id']);
            return ['success' => false, 'message' => 'Неверное имя пользователя или пароль'];
        }
        
        // Создание сессии
        $sessionToken = bin2hex(random_bytes(32));
        $expiresAt = date('Y-m-d H:i:s', strtotime('+1 hour'));
        
        $stmt = $this->db->prepare("INSERT INTO user_sessions (user_id, session_token, expires_at) VALUES (?, ?, ?)");
        $stmt->execute([$user['id'], $sessionToken, $expiresAt]);
        
        // Обновление времени последнего входа
        $stmt = $this->db->prepare("UPDATE users SET last_login = NOW() WHERE id = ?");
        $stmt->execute([$user['id']]);
        
        // Установка сессионных переменных
        $_SESSION['user_id'] = $user['id'];
        $_SESSION['session_token'] = $sessionToken;
        $_SESSION['username'] = $user['username'];
        
        return [
            'success' => true, 
            'message' => 'Вход выполнен успешно',
            'user' => [
                'id' => $user['id'],
                'username' => $user['username'],
                'email' => $user['email']
            ]
        ];
    }
    
    // Проверка прав доступа
    public function checkPermission($userId, $permissionName) {
        $stmt = $this->db->prepare("
            SELECT COUNT(*) as count
            FROM users u
            JOIN user_roles ur ON u.id = ur.user_id
            JOIN role_permissions rp ON ur.role_id = rp.role_id
            JOIN permissions p ON rp.permission_id = p.id
            WHERE u.id = ? AND u.is_active = TRUE AND p.name = ?
        ");
        
        $stmt->execute([$userId, $permissionName]);
        $result = $stmt->fetch(PDO::FETCH_ASSOC);
        
        return $result['count'] > 0;
    }
    
    // Получение ID роли
    private function getRoleId($roleName) {
        $stmt = $this->db->prepare("SELECT id FROM roles WHERE name = ?");
        $stmt->execute([$roleName]);
        $role = $stmt->fetch(PDO::FETCH_ASSOC);
        return $role['id'];
    }
    
    // Логирование неудачных попыток входа
    private function logFailedAttempt($userId) {
        $stmt = $this->db->prepare("
            INSERT INTO login_attempts (user_id, attempt_time, ip_address) 
            VALUES (?, NOW(), ?)
        ");
        $stmt->execute([$userId, $_SERVER['REMOTE_ADDR']]);
    }
}
?>
```

### **Задание 3: Middleware для проверки доступа**

**Файл: `middleware.php`**

```php
<?php
require_once 'auth.php';

class Middleware {
    private $auth;
    
    public function __construct($auth) {
        $this->auth = $auth;
    }
    
    // Проверка аутентификации
    public function requireAuth() {
        session_start();
        
        if (!isset($_SESSION['user_id']) || !isset($_SESSION['session_token'])) {
            header('HTTP/1.1 401 Unauthorized');
            echo json_encode(['error' => 'Требуется аутентификация']);
            exit;
        }
        
        // Проверка валидности сессии
        $stmt = $this->auth->db->prepare("
            SELECT * FROM user_sessions 
            WHERE user_id = ? AND session_token = ? AND expires_at > NOW()
        ");
        $stmt->execute([$_SESSION['user_id'], $_SESSION['session_token']]);
        
        if ($stmt->rowCount() === 0) {
            session_destroy();
            header('HTTP/1.1 401 Unauthorized');
            echo json_encode(['error' => 'Сессия истекла']);
            exit;
        }
        
        return $_SESSION['user_id'];
    }
    
    // Проверка роли
    public function requireRole($roleName) {
        $userId = $this->requireAuth();
        
        $stmt = $this->auth->db->prepare("
            SELECT COUNT(*) as count
            FROM users u
            JOIN user_roles ur ON u.id = ur.user_id
            JOIN roles r ON ur.role_id = r.id
            WHERE u.id = ? AND r.name = ?
        ");
        
        $stmt->execute([$userId, $roleName]);
        $result = $stmt->fetch(PDO::FETCH_ASSOC);
        
        if ($result['count'] === 0) {
            header('HTTP/1.1 403 Forbidden');
            echo json_encode(['error' => 'Недостаточно прав']);
            exit;
        }
        
        return $userId;
    }
    
    // Проверка разрешения
    public function requirePermission($permissionName) {
        $userId = $this->requireAuth();
        
        if (!$this->auth->checkPermission($userId, $permissionName)) {
            header('HTTP/1.1 403 Forbidden');
            echo json_encode(['error' => 'Недостаточно прав']);
            exit;
        }
        
        return $userId;
    }
}
?>
```

if ($_SERVER['REQUEST_METHOD'] === 'GET' && $_SERVER['REQUEST_URI'] === '/admin/users') {
    // Требуется роль администратора
    $userId = $middleware->requireRole('admin');
    
    // Получение списка пользователей
    $stmt = $db->prepare("
        SELECT u.id, u.username, u.email, u.created_at, u.last_login, 
               GROUP_CONCAT(r.name SEPARATOR ', ') as roles
        FROM users u
        LEFT JOIN user_roles ur ON u.id = ur.user_id
        LEFT JOIN roles r ON ur.role_id = r.id
        GROUP BY u.id
        ORDER BY u.created_at DESC

    $stmt->execute();
    $users = $stmt->fetchAll(PDO::FETCH_ASSOC);

### **Задание 2: Реализация регистрации и входа**

**Файл: `auth.php`**

```php
<?php
session_start();
require_once 'db.php';

class Auth {
    private $db;
    
    public function __construct($db) {
        $this->db = $db;
    }
    
    // Регистрация пользователя
    public function register($username, $email, $password) {
        // Валидация данных
        if (strlen($password) < 8) {
            return ['success' => false, 'message' => 'Пароль должен быть не менее 8 символов'];
        }
        
        // Хеширование пароля
        $passwordHash = password_hash($password, PASSWORD_DEFAULT);
        
        // Проверка существования пользователя
        $stmt = $this->db->prepare("SELECT id FROM users WHERE username = ? OR email = ?");
        $stmt->execute([$username, $email]);
        
        if ($stmt->rowCount() > 0) {
            return ['success' => false, 'message' => 'Пользователь с таким именем или email уже существует'];
        }
        
        // Создание пользователя
        $stmt = $this->db->prepare("INSERT INTO users (username, email, password_hash) VALUES (?, ?, ?)");
        $stmt->execute([$username, $email, $passwordHash]);
        
        // Назначение роли "user" по умолчанию
        $userId = $this->db->lastInsertId();
        $roleId = $this->getRoleId('user');
        
        $stmt = $this->db->prepare("INSERT INTO user_roles (user_id, role_id) VALUES (?, ?)");
        $stmt->execute([$userId, $roleId]);
        
        return ['success' => true, 'message' => 'Регистрация успешна'];
    }
    
    // Вход пользователя
    public function login($username, $password) {
        // Получение пользователя
        $stmt = $this->db->prepare("SELECT * FROM users WHERE username = ? AND is_active = TRUE");
        $stmt->execute([$username]);
        $user = $stmt->fetch(PDO::FETCH_ASSOC);
        
        if (!$user) {
            return ['success' => false, 'message' => 'Неверное имя пользователя или пароль'];
        }
        
        // Проверка пароля
        if (!password_verify($password, $user['password_hash'])) {
            // Логирование неудачной попытки
            $this->logFailedAttempt($user['id']);
            return ['success' => false, 'message' => 'Неверное имя пользователя или пароль'];
        }
        
        // Создание сессии
        $sessionToken = bin2hex(random_bytes(32));
        $expiresAt = date('Y-m-d H:i:s', strtotime('+1 hour'));
        
        $stmt = $this->db->prepare("INSERT INTO user_sessions (user_id, session_token, expires_at) VALUES (?, ?, ?)");
        $stmt->execute([$user['id'], $sessionToken, $expiresAt]);
        
        // Обновление времени последнего входа
        $stmt = $this->db->prepare("UPDATE users SET last_login = NOW() WHERE id = ?");
        $stmt->execute([$user['id']]);
        
        // Установка сессионных переменных
        $_SESSION['user_id'] = $user['id'];
        $_SESSION['session_token'] = $sessionToken;
        $_SESSION['username'] = $user['username'];
        
        return [
            'success' => true, 
            'message' => 'Вход выполнен успешно',
            'user' => [
                'id' => $user['id'],
                'username' => $user['username'],
                'email' => $user['email']
            ]
        ];
    }
    
    // Проверка прав доступа
    public function checkPermission($userId, $permissionName) {
        $stmt = $this->db->prepare("
            SELECT COUNT(*) as count
            FROM users u
            JOIN user_roles ur ON u.id = ur.user_id
            JOIN role_permissions rp ON ur.role_id = rp.role_id
            JOIN permissions p ON rp.permission_id = p.id
            WHERE u.id = ? AND u.is_active = TRUE AND p.name = ?
        ");
        
        $stmt->execute([$userId, $permissionName]);
        $result = $stmt->fetch(PDO::FETCH_ASSOC);
        
        return $result['count'] > 0;
    }
    
    // Получение ID роли
    private function getRoleId($roleName) {
        $stmt = $this->db->prepare("SELECT id FROM roles WHERE name = ?");
        $stmt->execute([$roleName]);
        $role = $stmt->fetch(PDO::FETCH_ASSOC);
        return $role['id'];
    }
    
    // Логирование неудачных попыток входа
    private function logFailedAttempt($userId) {
        $stmt = $this->db->prepare("
            INSERT INTO login_attempts (user_id, attempt_time, ip_address) 
            VALUES (?, NOW(), ?)
        ");
        $stmt->execute([$userId, $_SERVER['REMOTE_ADDR']]);
    }
}
?>
```

### **Задание 3: Middleware для проверки доступа**

**Файл: `middleware.php`**

```php
<?php
require_once 'auth.php';

class Middleware {
    private $auth;
    
    public function __construct($auth) {
        $this->auth = $auth;
    }
    
    // Проверка аутентификации
    public function requireAuth() {
        session_start();
        
        if (!isset($_SESSION['user_id']) || !isset($_SESSION['session_token'])) {
            header('HTTP/1.1 401 Unauthorized');
            echo json_encode(['error' => 'Требуется аутентификация']);
            exit;
        }
        
        // Проверка валидности сессии
        $stmt = $this->auth->db->prepare("
            SELECT * FROM user_sessions 
            WHERE user_id = ? AND session_token = ? AND expires_at > NOW()
        ");
        $stmt->execute([$_SESSION['user_id'], $_SESSION['session_token']]);
        
        if ($stmt->rowCount() === 0) {
            session_destroy();
            header('HTTP/1.1 401 Unauthorized');
            echo json_encode(['error' => 'Сессия истекла']);
            exit;
        }
        
        return $_SESSION['user_id'];
    }
    
    // Проверка роли
    public function requireRole($roleName) {
        $userId = $this->requireAuth();
        
        $stmt = $this->auth->db->prepare("
            SELECT COUNT(*) as count
            FROM users u
            JOIN user_roles ur ON u.id = ur.user_id
            JOIN roles r ON ur.role_id = r.id
            WHERE u.id = ? AND r.name = ?
        ");
        
        $stmt->execute([$userId, $roleName]);
        $result = $stmt->fetch(PDO::FETCH_ASSOC);
        
        if ($result['count'] === 0) {
            header('HTTP/1.1 403 Forbidden');
            echo json_encode(['error' => 'Недостаточно прав']);
            exit;
        }
        
        return $userId;
    }
    
    // Проверка разрешения
    public function requirePermission($permissionName) {
        $userId = $this->requireAuth();
        
        if (!$this->auth->checkPermission($userId, $permissionName)) {
            header('HTTP/1.1 403 Forbidden');
            echo json_encode(['error' => 'Недостаточно прав']);
            exit;
        }
        
        return $userId;
    }
}
?>
```
