```javascript
  },

    'init.sql': ''
  },

    'setup.js': '',
    'create-structure.js': ''
  },
  '.gitignore': '',
  'package.json': '',
  'README.md': '',
  'docker-compose.yml': ''
};

function createStructure(basePath, obj) {
  Object.entries(obj).forEach(([key, value]) => {
    const fullPath = path.join(basePath, key);
    
    if (typeof value === 'object') {
      if (!fs.existsSync(fullPath)) {
        fs.mkdirSync(fullPath, { recursive: true });
        console.log(`📁 Создана папка: ${fullPath}`);
      }
      
      if (Object.keys(value).length > 0) {
        createStructure(fullPath, value);
      }
    } else {
      fs.writeFileSync(fullPath, value);
      console.log(`📄 Создан файл: ${fullPath}`);
    }
  });
}

// Создаем структуру
createStructure(process.cwd(), structure);
console.log('✅ Структура проекта создана!');
```

### Запуск:
```bash
node scripts/create-structure.js
```

## 5. **Yeoman Generator (самый автоматизированный)**

### Создание своего генератора:

#### `generator-formdata-db/generators/app/index.js`:
```javascript
const Generator = require('yeoman-generator');

module.exports = class extends Generator {
  async prompting() {
    this.answers = await this.prompt([
      {
        type: 'input',
        name: 'projectName',
        message: 'Название проекта',
        default: 'formdata-to-db'
      },
      {
        type: 'checkbox',
        name: 'features',
        message: 'Выберите технологии',
        choices: [
          { name: 'MySQL', value: 'mysql', checked: true },
          { name: 'PostgreSQL', value: 'postgres', checked: false },
          { name: 'MongoDB', value: 'mongodb', checked: false },
          { name: 'TypeScript', value: 'typescript', checked: false },
          { name: 'Docker', value: 'docker', checked: true },
        ]
      }
    ]);
  }

  writing() {
    // Копирование структуры
    this.fs.copyTpl(
      this.templatePath('**/*'),
      this.destinationPath(this.answers.projectName),
      this.answers
    );

    // Создание package.json
    this.fs.copyTpl(
      this.templatePath('package.json'),
      this.destinationPath(this.answers.projectName, 'package.json'),
      this.answers
    );
  }

  install() {
    const projectPath = this.destinationPath(this.answers.projectName);
    this.spawnCommand('npm', ['install'], { cwd: projectPath });
  }
};
```

### Использование:

## Рекомендация:

Для вашего проекта **FormData-to-DB** я рекомендую использовать **Vite** потому что:

1. ⚡ **Молниеносная сборка** - HMR (Hot Module Replacement) за доли секунды
2. 🛠 **Простая настройка** - минимум конфигурации
3. 📦 **Встроенный dev сервер** с проксированием API
4. 🔧 **Готовые шаблоны** для Vanilla JS, React, Vue
5. 🚀 **Оптимизированная сборка** для production

### Быстрый старт с Vite:
```bash
# Создаем проект
npm create vite@latest formdata-to-db -- --template vanilla
cd formdata-to-db

# Устанавливаем backend зависимости
mkdir backend
cd backend
npm init -y
npm install express mysql2 multer cors dotenv

# Создаем структуру
node ../scripts/create-structure.js

# Запускаем всё
npm run dev
```

Этот подход позволяет автоматически создавать всю структуру проекта и быстро начать разработку без ручного создания папок и файлов!
