#### JavaScript (app.js)
```javascript
document.getElementById('userForm').addEventListener('submit', async function(e) {
    e.preventDefault();
    
    const form = e.target;
    const formData = new FormData(form);
    
    // Добавляем дополнительные данные
    formData.append('created_at', new Date().toISOString());
    
    // Показываем статус отправки
    const messageDiv = document.getElementById('message');
    messageDiv.textContent = 'Отправка данных...';
    messageDiv.style.color = 'blue';
    
    try {
        const response = await fetch('http://localhost:3000/api/users', {
            method: 'POST',
            body: formData
            // Не устанавливаем Content-Type вручную!
            // FormData сам установит multipart/form-data
        });
        
        const result = await response.json();
        
        if (response.ok) {
            messageDiv.textContent = `✅ Данные успешно сохранены! ID: ${result.userId}`;
            messageDiv.style.color = 'green';
            form.reset(); // Очищаем форму
        } else {
            messageDiv.textContent = `❌ Ошибка: ${result.error || 'Неизвестная ошибка'}`;
            messageDiv.style.color = 'red';
        }
    } catch (error) {
        console.error('Ошибка сети:', error);
        messageDiv.textContent = '❌ Ошибка сети. Проверьте подключение.';
        messageDiv.style.color = 'red';
    }
});

// Опционально: предпросмотр аватарки
document.getElementById('avatar').addEventListener('change', function(e) {
    const file = e.target.files[0];
    if (file) {
        const reader = new FileReader();
        reader.onload = function(e) {
            // Можно добавить предпросмотр изображения
            console.log('Файл выбран:', file.name, 'Размер:', file.size);
        };
        reader.readAsDataURL(file);
    }
});
```

### 2. Серверная часть (Backend)

#### Node.js + Express (server.js)
```javascript
const express = require('express');
const multer = require('multer');
const mysql = require('mysql2/promise');
const cors = require('cors');
const path = require('path');
const fs = require('fs').promises;

const app = express();
app.use(cors());
app.use(express.json());

// Настройка multer для загрузки файлов
const storage = multer.diskStorage({
    destination: function (req, file, cb) {
        cb(null, 'uploads/')
    },
    filename: function (req, file, cb) {
        const uniqueSuffix = Date.now() + '-' + Math.round(Math.random() * 1E9);
        cb(null, uniqueSuffix + path.extname(file.originalname))
    }
});

const upload = multer({ 
    storage: storage,
    limits: { fileSize: 5 * 1024 * 1024 }, // 5MB
    fileFilter: (req, file, cb) => {
        if (file.mimetype.startsWith('image/')) {
            cb(null, true);
        } else {
            cb(new Error('Только изображения разрешены!'), false);
        }
    }
});

// Подключение к MySQL
const pool = mysql.createPool({
    host: 'localhost',
    user: 'root',
    password: 'password',
    database: 'users_db',
    waitForConnections: true,
    connectionLimit: 10,
    queueLimit: 0
});

// Создание таблицы (если не существует)
async function createTable() {
    const connection = await pool.getConnection();
    try {
        await connection.query(`
            CREATE TABLE IF NOT EXISTS users (
                id INT AUTO_INCREMENT PRIMARY KEY,
                name VARCHAR(100) NOT NULL,
                email VARCHAR(100) UNIQUE NOT NULL,
                age INT,
                avatar_path VARCHAR(255),
                subscribe BOOLEAN DEFAULT FALSE,
                created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
                updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
            )
        `);
        console.log('Таблица users готова');
    } catch (error) {
        console.error('Ошибка создания таблицы:', error);
    } finally {
        connection.release();
    }
}

createTable();

// Роут для сохранения пользователя
app.post('/api/users', upload.single('avatar'), async (req, res) => {
    try {
        console.log('Получены данные:', req.body);
        console.log('Файл:', req.file);
        
        // Валидация данных
        const { name, email, age, subscribe, created_at } = req.body;
        
        if (!name || !email) {
            return res.status(400).json({ 
                error: 'Имя и email обязательны для заполнения' 
            });
        }
        
        // Подготовка данных для вставки
        const userData = {
            name: name.trim(),
            email: email.trim(),
            age: age ? parseInt(age) : null,
            subscribe: subscribe === 'true',
            avatar_path: req.file ? req.file.path : null,
            created_at: created_at || new Date()
        };
        
        // Вставка в базу данных
        const connection = await pool.getConnection();
        try {
            const [result] = await connection.query(
                'INSERT INTO users SET ?',
                userData
            );
            
            res.status(201).json({ 
                success: true,
                message: 'Пользователь успешно добавлен',
                userId: result.insertId,
                user: userData
            });
        } catch (dbError) {
            // Обработка ошибок базы данных
            if (dbError.code === 'ER_DUP_ENTRY') {
                res.status(409).json({ 
                    error: 'Пользователь с таким email уже существует' 
                });
            } else {
                throw dbError;
            }
        } finally {
            connection.release();
        }
        
    } catch (error) {
        console.error('Ошибка сервера:', error);
        res.status(500).json({ 
            error: 'Внутренняя ошибка сервера',
            details: process.env.NODE_ENV === 'development' ? error.message : undefined
        });
    }
});

// Статика для загруженных файлов
app.use('/uploads', express.static('uploads'));

// Получение списка пользователей (опционально)
app.get('/api/users', async (req, res) => {
    try {
        const [rows] = await pool.query('SELECT * FROM users ORDER BY created_at DESC');
        res.json(rows);
    } catch (error) {
        res.status(500).json({ error: 'Ошибка получения данных' });
    }
});

const PORT = process.env.PORT || 3000;
app.listen(PORT, () => {
    console.log(`Сервер запущен на порту ${PORT}`);
});
```

```javascript
// server-postgres.js
const { Pool } = require('pg');

const pool = new Pool({
    user: 'postgres',
    host: 'localhost',
    database: 'users_db',
    password: 'password',
    port: 5432,
});

// Роут для PostgreSQL
app.post('/api/users', upload.single('avatar'), async (req, res) => {
    try {
        const { name, email, age, subscribe } = req.body;
        
        const result = await pool.query(
            `INSERT INTO users (name, email, age, avatar_path, subscribe)
             VALUES ($1, $2, $3, $4, $5)
             RETURNING id, created_at`,
            [name, email, age, req.file?.path, subscribe === 'true']
        );
        
        res.status(201).json({
            success: true,
            userId: result.rows[0].id
        });
    } catch (error) {
        console.error('Ошибка:', error);
        res.status(500).json({ error: 'Ошибка сервера' });
    }
});
```

### 6. MongoDB вариант (используя mongoose)

```javascript
// server-mongodb.js
const mongoose = require('mongoose');
const express = require('express');
const multer = require('multer');

mongoose.connect('mongodb://localhost:27017/usersdb');

const userSchema = new mongoose.Schema({
    name: { type: String, required: true },
    email: { type: String, required: true, unique: true },
    age: Number,
    avatar: {
        data: Buffer,
        contentType: String,
        filename: String
    },
    subscribe: Boolean,
    created_at: { type: Date, default: Date.now }
});

const User = mongoose.model('User', userSchema);

app.post('/api/users', upload.single('avatar'), async (req, res) => {
    try {
        const user = new User({
            name: req.body.name,
            email: req.body.email,
            age: req.body.age,
            subscribe: req.body.subscribe === 'true',
            avatar: req.file ? {
                data: req.file.buffer, // Если используем memoryStorage
                contentType: req.file.mimetype,
                filename: req.file.originalname
            } : undefined
        });

        await user.save();
        res.status(201).json({ success: true, userId: user._id });
    } catch (error) {
        res.status(500).json({ error: error.message });
    }
});
```

### 7. Безопасность и валидация

```javascript
// Дополнительная валидация на сервере
function validateUserData(data) {
    const errors = [];
    
    // Валидация имени
    if (!data.name || data.name.length < 2) {
        errors.push('Имя должно содержать минимум 2 символа');
    }
    
    // Валидация email
    const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
    if (!emailRegex.test(data.email)) {
        errors.push('Некорректный email адрес');
    }
    
    // Валидация возраста
    if (data.age && (data.age < 1 || data.age > 120)) {
        errors.push('Возраст должен быть от 1 до 120 лет');
    }
    
    return errors;
}

// Добавить в роут
const validationErrors = validateUserData(req.body);
if (validationErrors.length > 0) {
    return res.status(400).json({ errors: validationErrors });
}
```

### Ключевые моменты:

1. **FormData** автоматически обрабатывает файлы и текстовые данные
2. **Multer** на сервере обрабатывает multipart/form-data
3. **Подготовленные SQL-запросы** предотвращают SQL-инъекции
4. **Валидация** на клиенте и сервере
5. **Обработка ошибок** на всех этапах
6. **CORS** для кросс-доменных запросов

Это полный пример отправки данных из формы через FormData и сохранения в базу данных MySQL/PostgreSQL/MongoDB.
