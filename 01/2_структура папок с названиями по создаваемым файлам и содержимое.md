### `frontend/js/app.js`
```javascript
// Конфигурация API
const API_BASE_URL = 'http://localhost:3000/api';

// DOM элементы
const userForm = document.getElementById('userForm');
const messageDiv = document.getElementById('message');
const avatarInput = document.getElementById('avatar');
const previewImage = document.getElementById('previewImage');
const imagePreview = document.getElementById('imagePreview');
const previewText = document.getElementById('previewText');
const fileName = document.getElementById('fileName');
const submitBtn = document.getElementById('submitBtn');
const resetBtn = document.getElementById('resetBtn');
const viewUsersBtn = document.getElementById('viewUsersBtn');
const clearHistoryBtn = document.getElementById('clearHistoryBtn');
const connectionStatus = document.getElementById('connectionStatus');
const connectionText = document.getElementById('connectionText');
const lastSend = document.getElementById('lastSend');
const usersModal = document.getElementById('usersModal');
const closeModal = document.getElementById('closeModal');
const usersList = document.getElementById('usersList');

// Состояние приложения
let isConnected = false;
let lastSendTime = null;

// Инициализация
document.addEventListener('DOMContentLoaded', () => {
    checkServerConnection();
    loadFromLocalStorage();
    setupEventListeners();
});

// Проверка соединения с сервером
async function checkServerConnection() {
    try {
        const response = await fetch(`${API_BASE_URL}/health`);
        if (response.ok) {
            setConnectionStatus(true);
        } else {
            setConnectionStatus(false);
        }
    } catch (error) {
        setConnectionStatus(false);
    }
}

// Установка статуса соединения
function setConnectionStatus(connected) {
    isConnected = connected;
    connectionStatus.className = `status-dot ${connected ? 'connected' : 'disconnected'}`;
    connectionText.textContent = connected ? 'Соединение установлено' : 'Сервер недоступен';
    submitBtn.disabled = !connected;
    submitBtn.title = connected ? 'Отправить форму' : 'Сервер недоступен';
}

// Настройка обработчиков событий
function setupEventListeners() {
    // Предпросмотр изображения
    avatarInput.addEventListener('change', handleImagePreview);
    
    // Сброс формы
    resetBtn.addEventListener('click', resetForm);
    
    // Отправка формы
    userForm.addEventListener('submit', handleFormSubmit);
    
    // Просмотр пользователей
    viewUsersBtn.addEventListener('click', showUsersList);
    
    // Очистка истории
    clearHistoryBtn.addEventListener('click', clearHistory);
    
    // Закрытие модального окна
    closeModal.addEventListener('click', () => {
        usersModal.classList.remove('active');
    });
    
    // Закрытие модального окна по клику вне его
    usersModal.addEventListener('click', (e) => {
        if (e.target === usersModal) {
            usersModal.classList.remove('active');
        }
    });
    
    // Периодическая проверка соединения
    setInterval(checkServerConnection, 30000);
}

// Предпросмотр выбранного изображения
function handleImagePreview(e) {
    const file = e.target.files[0];
    if (file) {
        // Обновляем имя файла
        fileName.textContent = file.name;
        
        // Показываем превью
        const reader = new FileReader();
        reader.onload = function(e) {
            previewImage.src = e.target.result;
            previewImage.style.display = 'block';
            previewText.style.display = 'none';
            imagePreview.classList.add('active');
        };
        reader.readAsDataURL(file);
        
        // Валидация размера файла
        const maxSize = 5 * 1024 * 1024; // 5MB
        if (file.size > maxSize) {
            showMessage(`Файл слишком большой (${(file.size / 1024 / 1024).toFixed(2)} MB). Максимум 5MB.`, 'error');
            avatarInput.value = '';
            imagePreview.classList.remove('active');
            fileName.textContent = 'Выберите файл...';
        }
    }
}

// Валидация формы
function validateForm(formData) {
    const errors = [];
    
    // Валидация имени
    const name = formData.get('name');
    if (!name || name.trim().length < 2) {
        errors.push('Имя должно содержать минимум 2 символа');
    }
    
    // Валидация email
    const email = formData.get('email');
    const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
    if (!email || !emailRegex.test(email)) {
        errors.push('Введите корректный email адрес');
    }
    
    // Валидация возраста
    const age = formData.get('age');
    if (age && (age < 1 || age > 120)) {
        errors.push('Возраст должен быть от 1 до 120 лет');
    }
    
    return errors;
}

// Обработка отправки формы
async function handleFormSubmit(e) {
    e.preventDefault();
    
    // Показываем индикатор загрузки
    submitBtn.innerHTML = '<i class="fas fa-spinner fa-spin"></i> Отправка...';
    submitBtn.disabled = true;
    
    // Создаем FormData
    const formData = new FormData(userForm);
    
    // Добавляем метаданные
    formData.append('timestamp', new Date().toISOString());
    formData.append('userAgent', navigator.userAgent);
    
    // Валидация
    const errors = validateForm(formData);
    if (errors.length > 0) {
        showMessage(errors.join('<br>'), 'error');
        submitBtn.innerHTML = '<i class="fas fa-paper-plane"></i> Отправить данные';
        submitBtn.disabled = false;
        return;
    }
    
    // Отправка данных
    try {
        const response = await fetch(`${API_BASE_URL}/users`, {
            method: 'POST',
            body: formData
        });
        
        const result = await response.json();
        
        if (response.ok) {
            // Успешная отправка
            showMessage(
                `✅ Пользователь успешно добавлен!<br>
                 ID: ${result.userId}<br>
                 Имя: ${result.user.name}<br>
                 Email: ${result.user.email}`,
                'success'
            );
            
            // Обновляем время последней отправки
            lastSendTime = new Date();
            lastSend.textContent = lastSendTime.toLocaleTimeString();
            saveToLocalStorage();
            
            // Очищаем форму через 2 секунды
            setTimeout(resetForm, 2000);
            
        } else {
            // Ошибка сервера
            showMessage(`❌ Ошибка: ${result.error || 'Неизвестная ошибка'}`, 'error');
        }
        
    } catch (error) {
        // Ошибка сети
        console.error('Ошибка сети:', error);
        showMessage('❌ Ошибка сети. Проверьте подключение к интернету.', 'error');
        
    } finally {
        // Восстанавливаем кнопку
        submitBtn.innerHTML = '<i class="fas fa-paper-plane"></i> Отправить данные';
        submitBtn.disabled = !isConnected;
    }
}

                    <span class="user-email">${user.email}</span>

function loadFromLocalStorage() {
    const saved = localStorage.getItem('formData');
    if (saved) {
        try {
            const data = JSON.parse(saved);
            document.getElementById('name').value = data.name || '';
            document.getElementById('email').value = data.email || '';
            document.getElementById('age').value = data.age || '';
            document.getElementById('bio').value = data.bio || '';

        }
    }
```

```javascript
const mysql = require('mysql2/promise');

class Database {
    constructor() {
        this.pool = null;
    }

    async connect() {
        try {
            this.pool = mysql.createPool({
                host: process.env.DB_HOST || 'localhost',
                user: process.env.DB_USER || 'root',
                password: process.env.DB_PASSWORD || 'password',
                database: process.env.DB_NAME || 'formdata_db',
                port: process.env.DB_PORT || 3306,
                waitForConnections: true,
                connectionLimit: 10,
                queueLimit: 0,
                enableKeepAlive: true,
                keepAliveInitialDelay: 0
            });

            // Проверка соединения
            const connection = await this.pool.getConnection();
            console.log('✅ База данных подключена успешно');
            connection.release();

            // Создание таблицы если не существует
            await this.createTables();
            
            return this.pool;
        } catch (error) {
            console.error('❌ Ошибка подключения к базе данных:', error.message);
            throw error;
        }
    }

    async createTables() {
        const createTableSQL = `
            CREATE TABLE IF NOT EXISTS users (
                id INT AUTO_INCREMENT PRIMARY KEY,
                name VARCHAR(100) NOT NULL,
                email VARCHAR(100) UNIQUE NOT NULL,
                age INT,
                bio TEXT,
                avatar_path VARCHAR(255),
                subscribe BOOLEAN DEFAULT FALSE,
                user_agent TEXT,
                created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
                updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
                INDEX idx_email (email),
                INDEX idx_created_at (created_at)
            ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;
        `;

        try {
            await this.pool.query(createTableSQL);
            console.log('✅ Таблица users проверена/создана');
        } catch (error) {
            console.error('❌ Ошибка создания таблицы:', error);
            throw error;
        }
    }

    async disconnect() {
        if (this.pool) {
            await this.pool.end();
            console.log('📴 Соединение с базой данных закрыто');
        }
    }

    getPool() {
        if (!this.pool) {
            throw new Error('База данных не подключена. Вызовите connect() сначала.');
        }
        return this.pool;
    }
}

module.exports = new Database();
```
