```python
# telegram_notifier.py
# Модуль для отправки уведомлений в Telegram

import requests
import os
from datetime import datetime

class TelegramNotifier:
    """Класс для отправки уведомлений в Telegram"""
    
    def __init__(self, bot_token=None, chat_id=None):
        """
        Инициализация уведомителя
        
        Args:
            bot_token: токен Telegram бота
            chat_id: ID чата для отправки уведомлений
        """
        self.bot_token = bot_token or os.environ.get('TELEGRAM_BOT_TOKEN')
        self.chat_id = chat_id or os.environ.get('TELEGRAM_CHAT_ID')
        self.enabled = bool(self.bot_token and self.chat_id)
        
        if self.enabled:
            self.api_url = f"https://api.telegram.org/bot{self.bot_token}/sendMessage"
            print("✅ Telegram уведомления включены")
        else:
            print("⚠️ Telegram уведомления отключены (не заданы токен или chat_id)")
    
    def send_message(self, message, parse_mode='HTML'):
        """
        Отправка сообщения в Telegram
        
        Args:
            message: текст сообщения
            parse_mode: форматирование ('HTML' или 'Markdown')
        
        Returns:
            bool: успешность отправки
        """
        if not self.enabled:
            return False
        
        try:
            payload = {
                'chat_id': self.chat_id,
                'text': message,
                'parse_mode': parse_mode
            }
            response = requests.post(self.api_url, json=payload, timeout=10)
            return response.status_code == 200
        except Exception as e:
            print(f"❌ Ошибка отправки в Telegram: {e}")
            return False
    
    def send_deployment_start(self, servers_count, os_type):
        """Уведомление о начале развертывания"""
        message = f"""
🚀 <b>НАЧАЛО РАЗВЕРТЫВАНИЯ</b>

📦 Количество серверов: {servers_count}
💻 Тип ОС: {os_type.upper()}
⏰ Время начала: {datetime.now().strftime('%Y-%m-%d %H:%M:%S')}

Статус: 🔄 В процессе...
        """
        return self.send_message(message.strip())
    
    def send_deployment_complete(self, results):
        """Уведомление о завершении развертывания"""
        total = len(results)
        success = sum(1 for r in results if r.get('success', False))
        failed = total - success
        
        # Формируем детали по серверам
        servers_details = ""
        for result in results:
            status_icon = "✅" if result.get('success', False) else "❌"
            servers_details += f"\n{status_icon} {result['server']}"
        
        message = f"""
🎯 <b>РАЗВЕРТЫВАНИЕ ЗАВЕРШЕНО</b>

📊 <b>СТАТИСТИКА:</b>
• Всего серверов: {total}
• ✅ Успешно: {success}
• ❌ Ошибок: {failed}
• 📈 Процент успеха: {success/total*100:.1f}%

⏰ Время завершения: {datetime.now().strftime('%Y-%m-%d %H:%M:%S')}

<b>ДЕТАЛИ:</b>{servers_details}
        """
        self.send_message(message.strip())
    
    def send_server_result(self, server_ip, success, duration, error_message=None):
        """Уведомление о результате на одном сервере"""
        status_icon = "✅" if success else "❌"
        status_text = "УСПЕШНО" if success else "ОШИБКА"
        
        message = f"""
{status_icon} <b>Сервер {server_ip}</b>

Статус: {status_text}
Время: {duration}с
{f'Ошибка: {error_message}' if error_message else ''}
        """
        self.send_message(message.strip())
    
    def send_error(self, error_message):
        """Уведомление об ошибке"""
        message = f"""
⚠️ <b>ОШИБКА В СИСТЕМЕ</b>

{error_message}

Время: {datetime.now().strftime('%Y-%m-%d %H:%M:%S')}
        """
        self.send_message(message.strip())
```

```python
class DeploymentThread(threading.Thread):
    """Поток для выполнения развертывания с уведомлениями"""
    
    def __init__(self, scripts_info, auth_method, auth_data, os_type):
        threading.Thread.__init__(self)
        self.scripts_info = scripts_info
        self.auth_method = auth_method
        self.auth_data = auth_data
        self.os_type = os_type
        self.deployer = None
    
    def run(self):
        global deployment_status
        deployment_status['is_running'] = True
        deployment_status['progress'] = 0
        deployment_status['results'] = []
        
        # Уведомление о начале
        notifier.send_deployment_start(len(self.scripts_info), self.os_type)
        
        # Создаём деплоер
        if self.auth_method == 'key':
            self.deployer = Deployer(ssh_key_path=self.auth_data)
        else:
            self.deployer = Deployer()
        
        total = len(self.scripts_info)
        
        for i, info in enumerate(self.scripts_info):
            deployment_status['current_server'] = info['ip']
            
            print(f"\n🚀 Деплой на {info['ip']}...")
            
            # Выполняем деплой
            try:
                if self.auth_method == 'password':
                    result = self.deployer.deploy_to_server(
                        server_ip=info['ip'],
                        local_script_path=info['script_path'],
                        username=info.get('user', 'root'),
                        password=self.auth_data
                    )
                else:
                    result = self.deployer.deploy_to_server(
                        server_ip=info['ip'],
                        local_script_path=info['script_path'],
                        username=info.get('user', 'root')
                    )
                
                deployment_status['results'].append(result)
                
                # Уведомление о результате каждого сервера
                notifier.send_server_result(
                    info['ip'], 
                    result.get('success', False),
                    result.get('duration', 0),
                    result.get('error', None)
                )
                
            except Exception as e:
                error_result = {
                    'server': info['ip'],
                    'success': False,
                    'error': str(e),
                    'duration': 0
                }
                deployment_status['results'].append(error_result)
                notifier.send_server_result(info['ip'], False, 0, str(e))
            
            deployment_status['progress'] = int(((i + 1) / total) * 100)
        
        deployment_status['is_running'] = False
        deployment_status['current_server'] = ''
        
        # Уведомление о завершении
        notifier.send_deployment_complete(deployment_status['results'])
```

### 1. Настройка Telegram бота

1. Напишите [@BotFather](https://t.me/BotFather) в Telegram
2. Создайте бота: `/newbot`
3. Получите **токен** (например: `1234567890:ABCdefGHIjklmNOPqrstUVwxyz`)
4. Получите **chat_id**:
   - Напишите боту любое сообщение
   - Перейдите по URL: `https://api.telegram.org/bot<ВАШ_ТОКЕН>/getUpdates`
   - Найдите `"chat":{"id":123456789}`

```bash
# Linux/Mac

docker_build.bat
```

```bash
# Установка зависимостей
```
