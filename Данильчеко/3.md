#!/bin/bash
# ============================================
# АВТОМАТИЧЕСКИ СГЕНЕРИРОВАННЫЙ СКРИПТ
# Сервер: {{ server_ip }}
# Дата генерации: {{ date }}
# ============================================

LOGFILE="/tmp/deploy_{{ server_ip | replace('.', '_') }}.log"

# Функция для логирования
log() {
    echo "[$(date '+%Y-%m-%d %H:%M:%S')] $1" | tee -a $LOGFILE
}

# Функция проверки успешности команды
check_success() {
    if [ $? -eq 0 ]; then
        log "✅ УСПЕШНО: $1"
        return 0
    else
        log "❌ ОШИБКА: $1"
        return 1
    fi
}

# Начало выполнения
log "========================================="
log "Начало развертывания на сервере {{ server_ip }}"
log "Пакеты для установки: {{ packages | join(', ') }}"
log "========================================="

# Обновление списка пакетов
log "Обновление списка пакетов..."
apt-get update -y >> $LOGFILE 2>&1
check_success "Обновление списка пакетов"

# Установка каждого пакета
{% for package in packages %}
log "Установка пакета: {{ package }}..."
apt-get install -y {{ package }} >> $LOGFILE 2>&1
check_success "Установка {{ package }}"
{% endfor %}

# Проверка установленных пакетов
log "========================================="
log "ПРОВЕРКА УСТАНОВКИ:"
{% for package in packages %}
if command -v {{ package }} >/dev/null 2>&1; then
    log "✅ {{ package }} - установлен и доступен"
else
    log "⚠️ {{ package }} - установлен, но не в PATH"
fi
{% endfor %}

log "========================================="
log "РАЗВЕРТЫВАНИЕ ЗАВЕРШЕНО"
log "Лог сохранен в: $LOGFILE"
log "========================================="

exit 0
```

# generator.py
# Модуль генерации bash-скриптов из шаблона

from jinja2 import Template
from datetime import datetime
import os
import yaml

class ScriptGenerator:
    """Класс для генерации скриптов развертывания"""
    
    def __init__(self, template_path='template.sh.j2', output_dir='generated_scripts'):
        """
        Инициализация генератора
        
        Args:
            template_path: путь к файлу шаблона
            output_dir: директория для сохранения сгенерированных скриптов
        """
        self.template_path = template_path
        self.output_dir = output_dir
        
        # Загружаем шаблон
        with open(template_path, 'r', encoding='utf-8') as f:
            self.template_content = f.read()
        
        self.template = Template(self.template_content)
        
        # Создаем директорию для скриптов, если её нет
        os.makedirs(output_dir, exist_ok=True)
    
    def generate_script(self, server_ip, packages):
        """
        Генерирует bash-скрипт для конкретного сервера
        
        Args:
            server_ip: IP-адрес сервера
            packages: список пакетов для установки
        
        Returns:
            str: путь к сгенерированному скрипту
        """
        # Подготавливаем данные для шаблона
        data = {
            'server_ip': server_ip,
            'packages': packages,
            'date': datetime.now().strftime("%Y-%m-%d %H:%M:%S")
        }
        
        # Генерируем содержимое скрипта
        script_content = self.template.render(**data)
        
        # Формируем имя файла
        safe_ip = server_ip.replace('.', '_')
        script_filename = os.path.join(self.output_dir, f'deploy_{safe_ip}.sh')
        
        # Сохраняем скрипт
        with open(script_filename, 'w', encoding='utf-8') as f:
            f.write(script_content)
        
        # Делаем скрипт исполняемым
        os.chmod(script_filename, 0o755)
        
        print(f"✅ Сгенерирован скрипт: {script_filename}")
        return script_filename
    
    def generate_all_from_config(self, config_path='config.yaml'):
        """
        Генерирует скрипты для всех серверов из конфигурационного файла
        
        Args:
            config_path: путь к YAML-файлу с конфигурацией
        
        Returns:
            list: список словарей с информацией о сгенерированных скриптах
        """
        # Загружаем конфигурацию
        with open(config_path, 'r', encoding='utf-8') as f:
            config = yaml.safe_load(f)
        
        scripts_info = []
        
        print("\n" + "="*60)
        print("ГЕНЕРАЦИЯ СКРИПТОВ")
        print("="*60)
        
        for server in config['servers']:
            server_ip = server['ip']
            packages = server['packages']
            
            print(f"\n📝 Генерация для сервера: {server_ip}")
            print(f"   Пакеты: {', '.join(packages)}")
            
            script_path = self.generate_script(server_ip, packages)
            
            scripts_info.append({
                'ip': server_ip,
                'user': server.get('user', 'root'),
                'script_path': script_path,
                'packages': packages
            })
        
        print("\n" + "="*60)
        print(f"✅ Сгенерировано {len(scripts_info)} скриптов")
        print("="*60)
        
        return scripts_info


# Для самостоятельного тестирования
if __name__ == "__main__":
    generator = ScriptGenerator()
    
    # Тестовая генерация
    test_script = generator.generate_script('192.168.1.100', ['nginx', 'curl'])
    print(f"\nТестовый скрипт создан: {test_script}")
    
    # Вывод содержимого для проверки
    print("\nСодержимое сгенерированного скрипта:")
    print("-"*40)
    with open(test_script, 'r') as f:
        print(f.read()[:500])  # Показываем первые 500 символов
    print("-"*40)
```

# deployer.py
# Модуль для развертывания скриптов на удаленных серверах

import subprocess
import time
import os
from datetime import datetime

class Deployer:
    """Класс для развертывания скриптов на удаленных серверах (через SSH)"""
    
    def __init__(self, ssh_key_path=None, timeout=60):
        """
        Инициализация деплоера
        
        Args:
            ssh_key_path: путь к приватному SSH-ключу (опционально)
            timeout: таймаут выполнения скрипта в секундах
        """
        self.ssh_key_path = ssh_key_path
        self.timeout = timeout
        self.results = []
    
    def _build_ssh_command(self, server_ip, command, username='root'):
        """
        Собирает SSH команду для выполнения на удаленном сервере
        
        Args:
            server_ip: IP сервера
            command: команда для выполнения
            username: имя пользователя
        
        Returns:
            str: готовая SSH команда
        """
        ssh_base = f"ssh -o StrictHostKeyChecking=no -o ConnectTimeout=10"
        
        if self.ssh_key_path and os.path.exists(self.ssh_key_path):
            ssh_base += f" -i {self.ssh_key_path}"
        
        return f"{ssh_base} {username}@{server_ip} '{command}'"
    
    def copy_script(self, server_ip, local_script_path, username='root'):
        """
        Копирует скрипт на удаленный сервер через scp
        
        Args:
            server_ip: IP сервера
            local_script_path: путь к локальному скрипту
            username: имя пользователя
        
        Returns:
            bool: успешность копирования
        """
        try:
            remote_path = f"/tmp/deploy_script_{int(time.time())}.sh"
            
            scp_base = f"scp -o StrictHostKeyChecking=no"
            
            if self.ssh_key_path and os.path.exists(self.ssh_key_path):
                scp_base += f" -i {self.ssh_key_path}"
            
            scp_command = f"{scp_base} {local_script_path} {username}@{server_ip}:{remote_path}"
            
            print(f"  📁 Копирование скрипта на {server_ip}...")
            result = subprocess.run(scp_command, shell=True, capture_output=True, text=True, timeout=30)
            
            if result.returncode == 0:
                print(f"  ✅ Скрипт скопирован: {remote_path}")
                return remote_path
            else:
                print(f"  ❌ Ошибка копирования: {result.stderr}")
                return None
                
        except Exception as e:
            print(f"  ❌ Исключение при копировании: {e}")
            return None
    
    def run_script(self, server_ip, remote_script_path, username='root'):
        """
        Запускает скрипт на удаленном сервере через SSH
        
        Args:
            server_ip: IP сервера
            remote_script_path: путь к скрипту на сервере
            username: имя пользователя
        
        Returns:
            dict: результат выполнения
        """
        try:
            print(f"  🚀 Запуск скрипта на {server_ip}...")
            
            # Делаем скрипт исполняемым и запускаем
            command = f"chmod +x {remote_script_path} && bash {remote_script_path}"
            ssh_command = self._build_ssh_command(server_ip, command, username)
            
            start_time = time.time()
            result = subprocess.run(ssh_command, shell=True, capture_output=True, text=True, timeout=self.timeout)
            end_time = time.time()
            
            return {
                'success': result.returncode == 0,
                'stdout': result.stdout,
                'stderr': result.stderr,
                'exit_code': result.returncode,
                'duration': round(end_time - start_time, 2)
            }
            
        except subprocess.TimeoutExpired:
            return {
                'success': False,
                'stdout': '',
                'stderr': 'Timeout exceeded',
                'exit_code': -1,
                'duration': self.timeout
            }
        except Exception as e:
            return {
                'success': False,
                'stdout': '',
                'stderr': str(e),
                'exit_code': -1,
                'duration': 0
            }
    
    def deploy_to_server(self, server_ip, local_script_path, username='root', password=None):
        """
        Полный цикл развертывания на одном сервере
        
        Args:
            server_ip: IP сервера
            local_script_path: путь к локальному скрипту
            username: имя пользователя
            password: пароль (если не используется SSH-ключ)
        
        Returns:
            dict: результат развертывания
        """
        print(f"\n{'='*50}")
        print(f"🎯 Деплой на сервер: {server_ip}")
        print(f"{'='*50}")
        
        # Проверяем существование скрипта
        if not os.path.exists(local_script_path):
            return {
                'server': server_ip,
                'success': False,
                'error': f"Скрипт не найден: {local_script_path}"
            }
        
        # Если передан пароль, используем sshpass (нужно установить: apt install sshpass)
        if password:
            # Для простоты, используем старый метод с передачей пароля
            # В реальном проекте лучше использовать SSH-ключи
            print(f"  ⚠️ Используется аутентификация по паролю (рекомендуется SSH-ключ)")
        
        # Копируем скрипт
        remote_path = self.copy_script(server_ip, local_script_path, username)
        if not remote_path:
            return {
                'server': server_ip,
                'success': False,
                'error': 'Не удалось скопировать скрипт'
            }
        
        # Запускаем скрипт
        result = self.run_script(server_ip, remote_path, username)
        
        # Выводим результат
        if result['success']:
            print(f"  ✅ Деплой на {server_ip} УСПЕШНО завершен за {result['duration']}с")
        else:
            print(f"  ❌ Деплой на {server_ip} завершен с ошибкой (код: {result['exit_code']})")
            if result['stderr']:
                print(f"  Ошибка: {result['stderr'][:200]}")
        
        # Выводим последние строки лога
        if result['stdout']:
            lines = result['stdout'].strip().split('\n')
            print(f"\n  📋 Последние строки вывода:")
            for line in lines[-5:]:  # Показываем последние 5 строк
                print(f"    {line}")
        
        return {
            'server': server_ip,
            'success': result['success'],
            'stdout': result['stdout'],
            'stderr': result['stderr'],
            'exit_code': result['exit_code'],
            'duration': result['duration']
        }
    
    def deploy_to_all(self, scripts_info, password=None):
        """
        Развертывание на всех серверах
        
        Args:
            scripts_info: список с информацией о серверах и скриптах
            password: пароль (если используется)
        
        Returns:
            list: результаты развертывания
        """
        self.results = []
        
        print("\n" + "="*60)
        print("🚀 НАЧАЛО РАЗВЕРТЫВАНИЯ")
        print("="*60)
        
        for info in scripts_info:
            result = self.deploy_to_server(
                server_ip=info['ip'],
                local_script_path=info['script_path'],
                username=info.get('user', 'root'),
                password=password
            )
            self.results.append(result)
        
        return self.results
    
    def print_summary(self):
        """Выводит сводку результатов развертывания"""
        print("\n" + "="*60)
        print("📊 ИТОГОВЫЙ ОТЧЕТ")
        print("="*60)
        
        success_count = sum(1 for r in self.results if r['success'])
        total_count = len(self.results)
        
        print(f"\nВсего серверов: {total_count}")
        print(f"✅ Успешно: {success_count}")
        print(f"❌ Ошибок: {total_count - success_count}")
        print(f"🎯 Процент успеха: {success_count/total_count*100:.1f}%")
        
        print(f"\n{'='*60}")
        print("ДЕТАЛИ ПО КАЖДОМУ СЕРВЕРУ:")
        print(f"{'='*60}")
        
        for result in self.results:
            status = "✅ УСПЕХ" if result['success'] else "❌ ОШИБКА"
            duration = result.get('duration', 0)
            print(f"{result['server']:20} {status:10} (время: {duration}с)")
            
            if not result['success'] and 'error' in result:
                print(f"  └─ Причина: {result['error'][:100]}")


# Для самостоятельного тестирования
if __name__ == "__main__":
    # Создаем тестовый скрипт
    test_script = "/tmp/test_deploy.sh"
    with open(test_script, 'w') as f:
        f.write("#!/bin/bash\necho 'Hello from remote server!'\nuptime")
    os.chmod(test_script, 0o755)
    
    # Создаем деплоер
    deployer = Deployer()
    
    # Тест на локальной машине (для проверки)
    print("Тест на локальной машине (эхо-команда):")
    
    # Для реального теста раскомментируйте строку ниже и укажите ваш сервер
    # result = deployer.deploy_to_server('192.168.1.10', test_script, 'root', password='your_password')
    # print(f"\nРезультат: {result}")
    
    print("\n⚠️ Для реального развертывания укажите IP сервера и пароль")

```python
# main.py
# ГЛАВНЫЙ ФАЙЛ - ЗАПУСК ВСЕЙ СИСТЕМЫ

from generator import ScriptGenerator
from deployer import Deployer
import sys
import os

def print_banner():
    """Выводит красивый баннер в начале работы"""
    banner = """
    ╔══════════════════════════════════════════════════════════════╗
    ║                                                              ║
    ║     СИСТЕМА АВТОМАТИЧЕСКОГО РАЗВЕРТЫВАНИЯ ПО                ║
    ║     ГЕНЕРАЦИЯ СЕРВЕРНЫХ СКРИПТОВ                             ║
    ║                                                              ║
    ║     Версия: 1.0                                             ║
    ║     Автор: Дипломный проект                                  ║
    ║                                                              ║
    ╚══════════════════════════════════════════════════════════════╝
    """
    print(banner)

def check_requirements():
    """Проверяет наличие необходимых файлов"""
    required_files = ['config.yaml', 'template.sh.j2']
    missing = []
    
    for file in required_files:
        if not os.path.exists(file):
            missing.append(file)
    
    if missing:
        print("\n❌ ОШИБКА: Отсутствуют необходимые файлы:")
        for file in missing:
            print(f"   - {file}")
        print("\nПожалуйста, создайте эти файлы перед запуском.")
        return False
    
    return True

def get_auth_method():
    """Запрашивает способ аутентификации у пользователя"""
    print("\n🔐 Выберите способ аутентификации:")
    print("   1. SSH-ключ (рекомендуется)")
    print("   2. Пароль")
    print("   3. Только генерация скриптов (без развертывания)")
    
    choice = input("\nВаш выбор (1/2/3): ").strip()
    
    if choice == '1':
        key_path = input("Путь к SSH-ключу (например: ~/.ssh/id_rsa): ").strip()
        key_path = os.path.expanduser(key_path)
        if os.path.exists(key_path):
            return {'method': 'key', 'key_path': key_path}
        else:
            print(f"⚠️ Ключ не найден: {key_path}. Используем пароль.")
            return {'method': 'password', 'password': input("Введите пароль: ")}
    elif choice == '2':
        return {'method': 'password', 'password': input("Введите пароль для всех серверов: ")}
    else:
        return {'method': 'none'}

def main():
    """Главная функция"""
    print_banner()
    
    # Проверяем наличие файлов
    if not check_requirements():
        sys.exit(1)
    
    print("\n" + "="*60)
    print("ШАГ 1: ГЕНЕРАЦИЯ СКРИПТОВ")
    print("="*60)
    
    # Создаем генератор и генерируем скрипты
    generator = ScriptGenerator()
    scripts_info = generator.generate_all_from_config('config.yaml')
    
    if not scripts_info:
        print("❌ Не удалось сгенерировать скрипты. Проверьте config.yaml")
        sys.exit(1)
    
    # Запрашиваем метод аутентификации
    auth = get_auth_method()
    
    if auth['method'] == 'none':
        print("\n✅ Генерация скриптов завершена. Скрипты сохранены в папке 'generated_scripts/'")
        print("\nЧтобы выполнить развертывание, запустите скрипты вручную:")
        for info in scripts_info:
            print(f"  scp {info['script_path']} {info['user']}@{info['ip']}:/tmp/ && ssh {info['user']}@{info['ip']} 'bash /tmp/$(basename {info['script_path']})'")
        sys.exit(0)
    
    print("\n" + "="*60)
    print("ШАГ 2: РАЗВЕРТЫВАНИЕ НА СЕРВЕРАХ")
    print("="*60)
    
    # Создаем деплоер
    if auth['method'] == 'key':
        deployer = Deployer(ssh_key_path=auth['key_path'])
        results = deployer.deploy_to_all(scripts_info)
    else:
        deployer = Deployer()
        results = deployer.deploy_to_all(scripts_info, password=auth['password'])
    
    # Выводим отчет
    deployer.print_summary()
    
    # Сохраняем отчет в файл
    report_filename = f"deploy_report_{datetime.now().strftime('%Y%m%d_%H%M%S')}.txt"
    with open(report_filename, 'w') as f:
        f.write("ОТЧЕТ О РАЗВЕРТЫВАНИИ\n")
        f.write("="*60 + "\n")
        for result in results:
            f.write(f"\nСервер: {result['server']}\n")
            f.write(f"Статус: {'УСПЕХ' if result['success'] else 'ОШИБКА'}\n")
            if 'duration' in result:
                f.write(f"Время: {result['duration']}с\n")
            if 'error' in result:
                f.write(f"Ошибка: {result['error']}\n")
    
    print(f"\n📄 Полный отчет сохранен в файл: {report_filename}")

# Добавляем импорт datetime для отчета
from datetime import datetime

if __name__ == "__main__":
    main()

```

# Запустите тестовый контейнер Ubuntu
docker run -d --name test-server -p 2222:22 ubuntu:22.04 sleep infinity

docker exec test-server apt update && apt install -y openssh-server sudo

python main.py

## Что говорить на защите (готовый текст)

> **"Моя система состоит из двух основных модулей.**
> 
> **Первый модуль — генератор скриптов.** Он читает YAML-конфигурацию, где описано, какие пакеты нужно установить на какие серверы. На основе этой конфигурации и шаблона на языке Jinja2 он генерирует готовые bash-скрипты для каждого сервера.
> 
> **Второй модуль — деплоер.** Он копирует сгенерированные скрипты на удаленные серверы через SCP и запускает их через SSH. Все логи собираются и сохраняются в отчет.
> 
> **Преимущества моего подхода:**
> 1. Не нужно вручную писать скрипты для каждого сервера
> 2. Можно легко изменить конфигурацию и перегенерировать скрипты
> 3. Все действия логируются и можно отследить, где произошла ошибка
> 4. Система не требует установки агентов на целевые серверы — нужен только SSH"
