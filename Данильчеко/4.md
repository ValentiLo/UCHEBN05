# web_app.py
# Веб-интерфейс для системы развертывания

from flask import Flask, render_template, request, jsonify, send_file
from generator import ScriptGenerator
from deployer import Deployer
import yaml
import os
import json
import threading
from datetime import datetime

app = Flask(__name__)

# Глобальные переменные для хранения состояния
deployment_status = {
    'is_running': False,
    'progress': 0,
    'results': [],
    'current_server': ''
}

class DeploymentThread(threading.Thread):
    """Поток для выполнения развертывания (чтобы не блокировать веб-интерфейс)"""
    
    def __init__(self, scripts_info, auth_method, auth_data):
        threading.Thread.__init__(self)
        self.scripts_info = scripts_info
        self.auth_method = auth_method
        self.auth_data = auth_data
        self.deployer = None
    
    def run(self):
        global deployment_status
        deployment_status['is_running'] = True
        deployment_status['progress'] = 0
        deployment_status['results'] = []
        
        # Создаём деплоер
        if self.auth_method == 'key':
            self.deployer = Deployer(ssh_key_path=self.auth_data)
        else:
            self.deployer = Deployer()
        
        total = len(self.scripts_info)
        
        for i, info in enumerate(self.scripts_info):
            deployment_status['current_server'] = info['ip']
            
            # Определяем ОС для выбора скрипта
            script_path = info['script_path']
            
            # Выполняем деплой
            if self.auth_method == 'password':
                result = self.deployer.deploy_to_server(
                    server_ip=info['ip'],
                    local_script_path=script_path,
                    username=info.get('user', 'root'),
                    password=self.auth_data
                )
            else:
                result = self.deployer.deploy_to_server(
                    server_ip=info['ip'],
                    local_script_path=script_path,
                    username=info.get('user', 'root')
                )
            
            deployment_status['results'].append(result)
            deployment_status['progress'] = int(((i + 1) / total) * 100)
        
        deployment_status['is_running'] = False
        deployment_status['current_server'] = ''

@app.route('/')
def index():
    """Главная страница"""
    return render_template('index.html')

@app.route('/api/generate', methods=['POST'])
def generate_scripts():
    """API для генерации скриптов"""
    try:
        generator = ScriptGenerator()
        
        # Определяем ОС
        os_type = request.json.get('os_type', 'linux')
        
        # Генерируем скрипты
        scripts_info = generator.generate_all_from_config('config.yaml', os_type=os_type)
        
        return jsonify({
            'success': True,
            'scripts': scripts_info,
            'message': f'Сгенерировано {len(scripts_info)} скриптов'
        })
    except Exception as e:
        return jsonify({'success': False, 'error': str(e)})

@app.route('/api/deploy', methods=['POST'])
def deploy_scripts():
    """API для запуска развертывания"""
    global deployment_status
    
    if deployment_status['is_running']:
        return jsonify({'success': False, 'error': 'Развертывание уже выполняется'})
    
    auth_method = request.json.get('auth_method', 'password')
    auth_data = request.json.get('auth_data', '')
    os_type = request.json.get('os_type', 'linux')
    
    # Сначала генерируем скрипты
    generator = ScriptGenerator()
    scripts_info = generator.generate_all_from_config('config.yaml', os_type=os_type)
    
    if not scripts_info:
        return jsonify({'success': False, 'error': 'Не удалось сгенерировать скрипты'})
    
    # Запускаем поток развертывания
    thread = DeploymentThread(scripts_info, auth_method, auth_data)
    thread.start()
    
    return jsonify({'success': True, 'message': 'Развертывание начато'})

@app.route('/api/status', methods=['GET'])
def get_status():
    """API для получения статуса развертывания"""
    global deployment_status
    return jsonify({
        'is_running': deployment_status['is_running'],
        'progress': deployment_status['progress'],
        'current_server': deployment_status['current_server'],
        'results': deployment_status['results']
    })

@app.route('/api/config', methods=['GET', 'POST'])
def manage_config():
    """API для просмотра/изменения конфигурации"""
    config_path = 'config.yaml'
    
    if request.method == 'GET':
        with open(config_path, 'r', encoding='utf-8') as f:
            config = yaml.safe_load(f)
        return jsonify(config)
    
    elif request.method == 'POST':
        try:
            new_config = request.json
            with open(config_path, 'w', encoding='utf-8') as f:
                yaml.dump(new_config, f, allow_unicode=True)
            return jsonify({'success': True, 'message': 'Конфигурация сохранена'})
        except Exception as e:
            return jsonify({'success': False, 'error': str(e)})

@app.route('/api/logs/<server_ip>', methods=['GET'])
def get_logs(server_ip):
    """API для получения логов развертывания"""
    log_pattern = f"deploy_{server_ip.replace('.', '_')}.log"
    log_path = os.path.join('generated_scripts', log_pattern)
    
    if os.path.exists(log_path):
        with open(log_path, 'r') as f:
            logs = f.read()
        return jsonify({'success': True, 'logs': logs})
    else:
        return jsonify({'success': False, 'error': 'Лог не найден'})

if __name__ == '__main__':
    # Создаём папку templates если её нет
    os.makedirs('templates', exist_ok=True)
    print("🌐 Запуск веб-интерфейса на http://localhost:5000")
    app.run(debug=True, host='0.0.0.0', port=5000)

        self.generator = ScriptGenerator(
            template_path=self.template_path,
            output_dir=os.path.join(self.test_dir, 'scripts')

            self.assertIn('Server: 192.168.1.1', content)

        """Тест генерации из конфигурации"""

        # Проверяем результаты

        # Проверяем, что файлы созданы
        self.assertTrue(os.path.exists(scripts_info[0]['script_path']))
        self.assertTrue(os.path.exists(scripts_info[1]['script_path']))

## Обновлённый `generator.py` (с поддержкой Windows)

```python
# generator.py (обновлённая версия с поддержкой Windows)
from jinja2 import Template
from datetime import datetime
import os
import yaml

class ScriptGenerator:
    """Класс для генерации скриптов развертывания (Linux/Windows)"""
    
    def __init__(self, template_dir='.', output_dir='generated_scripts'):
        """
        Инициализация генератора
        
        Args:
            template_dir: директория с шаблонами
            output_dir: директория для сохранения сгенерированных скриптов
        """
        self.template_dir = template_dir
        self.output_dir = output_dir
        self.templates = {
            'linux': 'template.sh.j2',
            'windows': 'template.ps1.j2'
        }
        
        # Загружаем шаблоны
        self.template_cache = {}
        for os_type, template_file in self.templates.items():
            template_path = os.path.join(template_dir, template_file)
            if os.path.exists(template_path):
                with open(template_path, 'r', encoding='utf-8') as f:
                    self.template_cache[os_type] = Template(f.read())
            else:
                print(f"⚠️ Шаблон {template_file} не найден")
                self.template_cache[os_type] = None
        
        # Создаём директорию для скриптов
        os.makedirs(output_dir, exist_ok=True)
    
    def generate_script(self, server_ip, packages, os_type='linux'):
        """
        Генерирует скрипт для конкретного сервера
        
        Args:
            server_ip: IP-адрес сервера
            packages: список пакетов для установки
            os_type: тип ОС ('linux' или 'windows')
        
        Returns:
            str: путь к сгенерированному скрипту
        """
        template = self.template_cache.get(os_type)
        if not template:
            raise ValueError(f"Шаблон для ОС {os_type} не найден")
        
        # Подготавливаем данные для шаблона
        data = {
            'server_ip': server_ip,
            'packages': packages,
            'date': datetime.now().strftime("%Y-%m-%d %H:%M:%S")
        }
        
        # Генерируем содержимое скрипта
        script_content = template.render(**data)
        
        # Формируем имя файла
        safe_ip = server_ip.replace('.', '_')
        extension = '.sh' if os_type == 'linux' else '.ps1'
        script_filename = os.path.join(self.output_dir, f'deploy_{safe_ip}{extension}')
        
        # Сохраняем скрипт
        with open(script_filename, 'w', encoding='utf-8') as f:
            f.write(script_content)
        
        # Делаем скрипт исполняемым (только для Linux)
        if os_type == 'linux':
            os.chmod(script_filename, 0o755)
        
        print(f"✅ Сгенерирован скрипт: {script_filename}")
        return script_filename
    
    def generate_all_from_config(self, config_path='config.yaml', os_type='linux'):
        """
        Генерирует скрипты для всех серверов из конфигурационного файла
        
        Args:
            config_path: путь к YAML-файлу с конфигурацией
            os_type: тип ОС ('linux' или 'windows')
        
        Returns:
            list: список словарей с информацией о сгенерированных скриптах
        """
        with open(config_path, 'r', encoding='utf-8') as f:
            config = yaml.safe_load(f)
        
        scripts_info = []
        
        print("\n" + "="*60)
        print(f"ГЕНЕРАЦИЯ СКРИПТОВ (ОС: {os_type.upper()})")
        print("="*60)
        
        for server in config['servers']:
            server_ip = server['ip']
            packages = server['packages']
            
            print(f"\n📝 Генерация для сервера: {server_ip}")
            print(f"   Пакеты: {', '.join(packages)}")
            
            script_path = self.generate_script(server_ip, packages, os_type)
            
            scripts_info.append({
                'ip': server_ip,
                'user': server.get('user', 'root'),
                'script_path': script_path,
                'packages': packages,
                'os_type': os_type
            })
        
        print("\n" + "="*60)
        print(f"✅ Сгенерировано {len(scripts_info)} скриптов")
        print("="*60)
        
        return scripts_info
```
