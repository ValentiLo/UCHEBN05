```python
# metrics.py
# Prometheus метрики для системы развертывания

from prometheus_client import Counter, Gauge, Histogram, Info, generate_latest, CollectorRegistry
import time
from datetime import datetime

# Создаём регистр метрик
registry = CollectorRegistry()

# ============================================
# COUNTERS - счётчики событий
# ============================================

# Счётчик запущенных деплоев
deploys_total = Counter(
    'deploys_total',
    'Общее количество запусков развертывания',
    ['status', 'os_type'],
    registry=registry
)

# Счётчик сгенерированных скриптов
scripts_generated_total = Counter(
    'scripts_generated_total',
    'Общее количество сгенерированных скриптов',
    ['os_type'],
    registry=registry
)

# Счётчик ошибок
errors_total = Counter(
    'errors_total',
    'Общее количество ошибок',
    ['error_type', 'component'],
    registry=registry
)

# Счётчик успешных операций на серверах
server_operations_total = Counter(
    'server_operations_total',
    'Количество операций на серверах',
    ['server_ip', 'status'],
    registry=registry
)

# ============================================
# GAUGES - текущие значения
# ============================================

# Текущий статус деплоя (0 - idle, 1 - running)
deployment_status_gauge = Gauge(
    'deployment_status',
    'Текущий статус развертывания (0=idle, 1=running)',
    registry=registry
)

# Прогресс текущего деплоя (0-100)
deployment_progress = Gauge(
    'deployment_progress',
    'Прогресс текущего развертывания в процентах',
    registry=registry
)

# Количество активных серверов в конфигурации
active_servers_count = Gauge(
    'active_servers_count',
    'Количество серверов в конфигурации',
    registry=registry
)

# Время последнего успешного деплоя (unix timestamp)
last_successful_deploy = Gauge(
    'last_successful_deploy_timestamp',
    'Время последнего успешного развертывания',

    'Длительность генерации скриптов',

    registry=registry

class MetricsCollector:
    """Класс для сбора и обновления метрик"""
    
    def __init__(self):
        self.current_deploy_start_time = None
        self.current_generation_start_time = None
    
    def deploy_started(self, os_type='linux', servers_count=0):
        """Деплой начался"""
        deployment_status_gauge.set(1)
        self.current_deploy_start_time = time.time()
        deploys_total.labels(status='started', os_type=os_type).inc()
        active_servers_count.set(servers_count)
    
    def deploy_finished(self, success_count, fail_count, os_type='linux', duration=None):
        """Деплой завершился"""
        deployment_status_gauge.set(0)
        deployment_progress.set(100)
        
        if duration is None and self.current_deploy_start_time:
            duration = time.time() - self.current_deploy_start_time
        
        if duration:
            deploy_duration_seconds.observe(duration)
        
        if success_count > 0 and fail_count == 0:
            deploys_total.labels(status='success', os_type=os_type).inc()
            last_successful_deploy.set(time.time())
        else:
            deploys_total.labels(status='partial_success', os_type=os_type).inc()
        
        self.current_deploy_start_time = None
    
    def deploy_failed(self, error_type, os_type='linux'):
        """Деплой завершился ошибкой"""
        deployment_status_gauge.set(0)
        deploys_total.labels(status='failed', os_type=os_type).inc()
        errors_total.labels(error_type=error_type, component='deployer').inc()
        self.current_deploy_start_time = None
    
    def generation_started(self):
        """Началась генерация скриптов"""
        self.current_generation_start_time = time.time()
    
    def generation_finished(self, scripts_count, os_type='linux'):
        """Генерация скриптов завершилась"""
        if self.current_generation_start_time:
            duration = time.time() - self.current_generation_start_time
            script_generation_duration.observe(duration)
        
        scripts_generated_total.labels(os_type=os_type).inc(scripts_count)
        self.current_generation_start_time = None
    
    def generation_failed(self, error_type):
        """Генерация скриптов завершилась ошибкой"""
        errors_total.labels(error_type=error_type, component='generator').inc()
        self.current_generation_start_time = None
    
    def server_operation_result(self, server_ip, success, duration):
        """Результат операции на сервере"""
        status = 'success' if success else 'failed'
        server_operations_total.labels(server_ip=server_ip, status=status).inc()
    
    def update_progress(self, progress):
        """Обновление прогресса деплоя"""
        deployment_progress.set(progress)
    
    def update_packages_stats(self, packages_list):
        """Обновление статистики по пакетам"""
        packages_per_server.observe(len(packages_list))
    
    def record_error(self, error_type, component, message=''):
        """Запись ошибки"""
        errors_total.labels(error_type=error_type, component=component).inc()
        print(f"⚠️ Метрика ошибки: [{component}] {error_type} - {message}")


# Создаём глобальный экземпляр коллектора
metrics = MetricsCollector()

# Функция для получения метрик в формате Prometheus
def get_metrics():
    """Возвращает метрики в формате Prometheus"""
    return generate_latest(registry)


# Декоратор для автоматического замера времени
def timed(metric_name):
    """Декоратор для замера времени выполнения функции"""
    def decorator(func):
        def wrapper(*args, **kwargs):
            start = time.time()
            try:
                result = func(*args, **kwargs)
                duration = time.time() - start
                # Здесь можно добавить запись в метрику
                return result
            except Exception as e:
                metrics.record_error(str(type(e).__name__), metric_name)
                raise
        return wrapper
    return decorator


# Функция для health check (для Kubernetes)
def health_check():
    """Проверка работоспособности системы"""
    return {
        'status': 'healthy',
        'timestamp': datetime.now().isoformat(),
        'metrics': {
            'total_deploys': deploys_total._value.get(),
            'current_status': 'running' if deployment_status_gauge._value.get() else 'idle'
        }
    }

if __name__ == "__main__":
    # Тестовый вывод метрик
    print("Prometheus метрики:")
    print(get_metrics().decode('utf-8'))
```

```python
# web_app.py (добавлен эндпоинт /metrics для Prometheus)
from flask import Flask, render_template, request, jsonify, Response
from generator import ScriptGenerator
from deployer import Deployer
from telegram_notifier import TelegramNotifier
from metrics import metrics, get_metrics, health_check
import yaml
import os
import threading
from datetime import datetime

app = Flask(__name__)

# Инициализация уведомителя
notifier = TelegramNotifier()

# Глобальные переменные для хранения состояния
deployment_status = {
    'is_running': False,
    'progress': 0,
    'results': [],
    'current_server': ''
}

class DeploymentThread(threading.Thread):
    """Поток для выполнения развертывания с метриками"""
    
    def __init__(self, scripts_info, auth_method, auth_data, os_type):
        threading.Thread.__init__(self)
        self.scripts_info = scripts_info
        self.auth_method = auth_method
        self.auth_data = auth_data
        self.os_type = os_type
        self.deployer = None
        self.start_time = None
    
    def run(self):
        global deployment_status
        
        metrics.deploy_started(self.os_type, len(self.scripts_info))
        notifier.send_deployment_start(len(self.scripts_info), self.os_type)
        
        deployment_status['is_running'] = True
        deployment_status['progress'] = 0
        deployment_status['results'] = []
        
        if self.auth_method == 'key':
            self.deployer = Deployer(ssh_key_path=self.auth_data)
        else:
            self.deployer = Deployer()
        
        total = len(self.scripts_info)
        success_count = 0
        fail_count = 0
        
        for i, info in enumerate(self.scripts_info):
            deployment_status['current_server'] = info['ip']
            metrics.update_progress(int(((i) / total) * 100))
            
            start_time = datetime.now()
            
            try:
                if self.auth_method == 'password':
                    result = self.deployer.deploy_to_server(
                        server_ip=info['ip'],
                        local_script_path=info['script_path'],
                        username=info.get('user', 'root'),
                        password=self.auth_data
                    )
                else:
                    result = self.deployer.deploy_to_server(
                        server_ip=info['ip'],
                        local_script_path=info['script_path'],
                        username=info.get('user', 'root')
                    )
                
                duration = (datetime.now() - start_time).total_seconds()
                metrics.server_operation_result(info['ip'], result.get('success', False), duration)
                
                if result.get('success', False):
                    success_count += 1
                else:
                    fail_count += 1
                
                deployment_status['results'].append(result)
                notifier.send_server_result(info['ip'], result.get('success', False), duration)
                
            except Exception as e:
                fail_count += 1
                error_result = {
                    'server': info['ip'],
                    'success': False,
                    'error': str(e),
                    'duration': 0
                }
                deployment_status['results'].append(error_result)
                metrics.record_error(str(type(e).__name__), 'deployer', str(e))
                notifier.send_server_result(info['ip'], False, 0, str(e))
            
            metrics.update_progress(int(((i + 1) / total) * 100))
        
        deployment_status['is_running'] = False
        deployment_status['current_server'] = ''
        
        duration = (datetime.now() - self.start_time).total_seconds() if self.start_time else 0
        metrics.deploy_finished(success_count, fail_count, self.os_type, duration)
        notifier.send_deployment_complete(deployment_status['results'])
    
    def start(self):
        self.start_time = datetime.now()
        super().start()

@app.route('/api/deploy', methods=['POST'])
def deploy_scripts():
    global deployment_status
    
    if deployment_status['is_running']:
        return jsonify({'success': False, 'error': 'Развертывание уже выполняется'})
    
    auth_method = request.json.get('auth_method', 'password')
    auth_data = request.json.get('auth_data', '')
    os_type = request.json.get('os_type', 'linux')
    
    try:
        generator = ScriptGenerator()
        scripts_info = generator.generate_all_from_config('config.yaml', os_type=os_type)
        
        if not scripts_info:
            return jsonify({'success': False, 'error': 'Не удалось сгенерировать скрипты'})
    except Exception as e:
        metrics.record_error(str(type(e).__name__), 'generator', str(e))
        return jsonify({'success': False, 'error': str(e)})
    
    thread = DeploymentThread(scripts_info, auth_method, auth_data, os_type)
    thread.start()
    
    return jsonify({'success': True, 'message': 'Развертывание начато'})
```
