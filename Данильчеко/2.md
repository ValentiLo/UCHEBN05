 Для диплома **не нужен супер-сложный код** — нужна **понятная архитектура**, которая демонстрирует **саму идею**: генерация скриптов → доставка → выполнение.

**максимально простой, но рабочий код** на Python, который можно объяснить на защите за 2 минуты.

---

## Общая концепция (для презентации)

```
Пользователь → config.yaml → ГЕНЕРАТОР → script.sh → SSH → Удаленная машина → Лог
```

**Три простых шага:**
1. Читаем YAML-файл с описанием того, что нужно установить
2. Генерируем bash-скрипт по шаблону
3. Копируем и запускаем скрипт на удаленной машине через SSH

echo "=== Установка завершена на {{ server_ip }} ===" | tee -a $LOGFILE

## Шаг 3. Генератор скриптов `generator.py`

```python
# generator.py
from jinja2 import Template
from datetime import datetime
import os

def generate_script(server_ip, packages):
    """Генерирует bash-скрипт из шаблона"""
    
    # Читаем шаблон
    with open('template.sh.j2', 'r') as f:
        template_content = f.read()
    
    # Создаем шаблон Jinja2
    template = Template(template_content)
    
    # Подставляем переменные
    script_content = template.render(
        server_ip=server_ip,
        packages=packages,
        date=datetime.now().strftime("%Y-%m-%d %H:%M:%S")
    )
    
    # Сохраняем сгенерированный скрипт
    script_filename = f"generated_scripts/deploy_{server_ip}.sh"
    os.makedirs('generated_scripts', exist_ok=True)
    
    with open(script_filename, 'w') as f:
        f.write(script_content)
    
    # Делаем скрипт исполняемым
    os.chmod(script_filename, 0o755)
    
    print(f"✅ Сгенерирован скрипт: {script_filename}")
    return script_filename

# Пример использования
if __name__ == "__main__":
    generate_script("192.168.1.10", ["nginx", "curl"])
```

## Шаг 4. Деплоер (отправка и запуск) `deployer.py`

```python
# deployer.py
import paramiko
import time

def run_script_on_server(server_ip, username, script_path, password=None, key_path=None):
    """
    Копирует и запускает скрипт на удаленном сервере
    """
    print(f"\n🚀 Начинаем деплой на {server_ip}")
    
    # Создаем SSH клиент
    ssh = paramiko.SSHClient()
    ssh.set_missing_host_key_policy(paramiko.AutoAddPolicy())
    
    try:
        # Подключаемся
        if key_path:
            ssh.connect(server_ip, username=username, key_filename=key_path, timeout=10)
        else:
            ssh.connect(server_ip, username=username, password=password, timeout=10)
        
        # Копируем скрипт на сервер
        remote_script_path = f"/tmp/deploy_{server_ip}.sh"
        sftp = ssh.open_sftp()
        sftp.put(script_path, remote_script_path)
        sftp.close()
        print(f"  📁 Скрипт скопирован: {remote_script_path}")
        
        # Запускаем скрипт
        stdin, stdout, stderr = ssh.exec_command(f"bash {remote_script_path}")
        
        # Читаем вывод в реальном времени
        print(f"  📋 Выполнение скрипта:\n{'-'*40}")
        for line in stdout:
            print(f"    {line.strip()}")
        
        # Проверяем ошибки
        errors = stderr.read().decode()
        if errors:
            print(f"  ❌ Ошибки:\n{errors}")
        
        # Получаем код возврата
        exit_status = stdout.channel.recv_exit_status()
        
        if exit_status == 0:
            print(f"  ✅ Деплой на {server_ip} УСПЕШНО завершен")
        else:
            print(f"  ❌ Деплой на {server_ip} завершен с ошибкой (код: {exit_status})")
        
        return exit_status == 0
        
    except Exception as e:
        print(f"  ❌ Ошибка подключения к {server_ip}: {e}")
        return False
    finally:
        ssh.close()

# Пример использования
if __name__ == "__main__":
    run_script_on_server("192.168.1.10", "root", "generated_scripts/deploy_192.168.1.10.sh", password="your_password")
```

## Шаг 5. Главный файл `main.py` (всё вместе)

```python
# main.py
import yaml
from generator import generate_script
from deployer import run_script_on_server

def main():
    # 1. Загружаем конфигурацию
    with open('config.yaml', 'r') as f:
        config = yaml.safe_load(f)
    
    results = []
    
    # 2. Для каждого сервера
    for server in config['servers']:
        server_ip = server['ip']
        username = server.get('user', 'root')
        packages = server['packages']
        
        print(f"\n{'='*60}")
        print(f"Обработка сервера: {server_ip}")
        print(f"Пакеты: {', '.join(packages)}")
        
        # 3. Генерируем скрипт
        script_path = generate_script(server_ip, packages)
        
        # 4. Запускаем на удаленном сервере
        success = run_script_on_server(
            server_ip=server_ip,
            username=username,
            script_path=script_path,
            password=input(f"Введите пароль для {server_ip}: "),
            # или используйте SSH ключ: key_path="/home/user/.ssh/id_rsa"
        )
        
        results.append({
            'server': server_ip,
            'success': success
        })
    
    # 5. Выводим итоговый отчет
    print(f"\n{'='*60}")
    print("ИТОГОВЫЙ ОТЧЕТ")
    for result in results:
        status = "✅ УСПЕХ" if result['success'] else "❌ ОШИБКА"
        print(f"{result['server']}: {status}")

if __name__ "__main__":
    main()
```

## Как запустить (инструкция для диплома)

### Установка зависимостей
```bash
pip install paramiko pyyaml jinja2
```

### Подготовка
1. Установите SSH-сервер на целевой машине: `sudo apt install openssh-server`
2. Убедитесь, что можете подключиться: `ssh root@192.168.1.10`

### Запуск
```bash
python main.py
```

## Самое главное для защиты (ваши слова)

> **"Моя система решает проблему ручного развертывания ПО на многих серверах. Вместо того чтобы заходить на каждый сервер и вручную писать apt install, я описываю в YAML-файле, что нужно поставить, и система автоматически генерирует bash-скрипт через шаблонизатор Jinja2, копирует его на сервер по SSH и выполняет. При этом все логи собираются в одном месте."**

- **Генерация скриптов** (вместо ручного написания)
- **Шаблонизация** (Jinja2)
- **Автоматизация** (SSH, Paramiko)
- **Идемпотентность** (повторный запуск не ломает систему)

## Как упростить еще больше (если сложно с Paramiko)

Вместо Paramiko можно использовать **простой scp + ssh через subprocess**:

```python
import subprocess

def run_script_simple(server_ip, script_path):
    # Копируем через scp
    subprocess.run(f"scp {script_path} root@{server_ip}:/tmp/script.sh", shell=True)
    
    # Запускаем через ssh
    result = subprocess.run(f"ssh root@{server_ip} 'bash /tmp/script.sh'", shell=True, capture_output=True, text=True)
    
    print(result.stdout)
    return result.returncode == 0
```

Это **еще проще** и не требует установки Paramiko!
