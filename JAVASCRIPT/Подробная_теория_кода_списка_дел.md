## script5.js
```
// Получаем элементы
const form = document.querySelector('.add-form');
const input = document.querySelector('.add-form-input');
const todoList = document.querySelector('.todo-list');
const emptyMessage = document.querySelector('.empty-tasks');
const template = document.getElementById('task-template');

// Функция обновления сообщения о пустом списке
function updateEmptyMessage() {
    const items = document.querySelectorAll('.todo-list-item');
    if (items.length === 0) {
        emptyMessage.classList.remove('hidden');
    } else {
        emptyMessage.classList.add('hidden');
    }
}

// Функция добавления новой задачи
function addTask(taskText) {
    // Клонируем шаблон
    const newTask = template.content.cloneNode(true);
    const taskItem = newTask.querySelector('.todo-list-item');
    const taskSpan = newTask.querySelector('span');
    const taskCheckbox = newTask.querySelector('.todo-list-input');
    
    // Заполняем текст задачи
    taskSpan.textContent = taskText;
    
    // Добавляем обработчик для удаления при нажатии на чекбокс (опционально)
    taskCheckbox.addEventListener('change', function() {
        if (this.checked) {
            taskItem.remove();
            updateEmptyMessage();
        }
    });
    
    // Добавляем задачу в список
    todoList.appendChild(taskItem);
    updateEmptyMessage();
}

// Обработчик отправки формы
form.addEventListener('submit', function(event) {
    event.preventDefault(); // Отменяем отправку на сервер
    
    const taskText = input.value.trim();
    
    if (taskText !== '') {
        addTask(taskText);
        input.value = ''; // Очищаем поле ввода
        input.focus(); // Возвращаем фокус на поле
    }
});

// Добавляем обработчики для существующих задач (чтобы удалялись при нажатии)
document.querySelectorAll('.todo-list-item .todo-list-input').forEach(checkbox => {
    checkbox.addEventListener('change', function() {
        if (this.checked) {
            this.closest('.todo-list-item').remove();
            updateEmptyMessage();
        }
    });
});

// Проверяем пустой список при загрузке
updateEmptyMessage();


```

## 1. **Получение элементов DOM**

```javascript
const form = document.querySelector('.add-form');
const input = document.querySelector('.add-form-input');
const todoList = document.querySelector('.todo-list');
const emptyMessage = document.querySelector('.empty-tasks');
const template = document.getElementById('task-template');
```

### 🔍 **Что такое DOM?**
**DOM (Document Object Model)** — это представление HTML-документа в виде дерева объектов, с которым JavaScript может взаимодействовать.

### 📖 **Методы поиска элементов:**

| Метод | Что ищет | Возвращает | Пример |
|-------|----------|------------|--------|
| `querySelector('.класс')` | Первый элемент с классом | 1 элемент | `document.querySelector('.add-form')` |
| `querySelectorAll('.класс')` | Все элементы с классом | Коллекция элементов | `document.querySelectorAll('.todo-list-item')` |
| `getElementById('id')` | Элемент с ID | 1 элемент | `document.getElementById('task-template')` |

### 💡 **Пример для понимания:**
```javascript
// HTML: <div class="my-class">Привет</div>
const div = document.querySelector('.my-class');
console.log(div.textContent); // Выведет: "Привет"
```

## 2. **Функция `updateEmptyMessage()`**

```javascript
function updateEmptyMessage() {
    const items = document.querySelectorAll('.todo-list-item');
    if (items.length === 0) {
        emptyMessage.classList.remove('hidden');
    } else {
        emptyMessage.classList.add('hidden');
    }
}
```

### 🔍 **Пошаговый разбор:**

1. **`querySelectorAll('.todo-list-item')`** — находит ВСЕ элементы с классом `todo-list-item` (все задачи)
2. **`.length`** — свойство, показывающее количество найденных элементов
3. **`classList`** — специальный объект для работы с классами элемента:
   - `add('hidden')` — добавляет класс (скрывает сообщение)
   - `remove('hidden')` — удаляет класс (показывает сообщение)
   - `toggle('hidden')` — переключает (есть/нет)

### 📊 **Визуализация условия:**
```
items.length === 0  →  задач нет  →  показываем сообщение
items.length > 0    →  есть задачи →  скрываем сообщение
```

### 💡 **Простой пример:**
```javascript
// Допустим, есть 3 задачи
const tasks = document.querySelectorAll('.task');
console.log(tasks.length); // 3
if (tasks.length === 0) {
    console.log('Задач нет');
} else {
    console.log(`Найдено ${tasks.length} задач(и)`);
}
```

## 3. **Функция `addTask(taskText)`**

```javascript
function addTask(taskText) {
    const newTask = template.content.cloneNode(true);
    const taskItem = newTask.querySelector('.todo-list-item');
    const taskSpan = newTask.querySelector('span');
    const taskCheckbox = newTask.querySelector('.todo-list-input');
    
    taskSpan.textContent = taskText;
    
    taskCheckbox.addEventListener('change', function() {
        if (this.checked) {
            taskItem.remove();
            updateEmptyMessage();
        }
    });
    
    todoList.appendChild(taskItem);
    updateEmptyMessage();
}
```

### 🔍 **Что такое `<template>`?**

```html
<template id="task-template">
  <li class="todo-list-item">
    <label>
      <input type="checkbox" class="todo-list-input">
      <span></span>
    </label>
  </li>
</template>
```

**`<template>`** — это HTML-элемент, который:
- ❌ Не отображается на странице
- ✅ Хранит разметку "про запас"
- ✅ Используется для клонирования через JavaScript

### 📖 **Метод `cloneNode(true)`**

```javascript
template.content.cloneNode(true)
```

| Часть | Значение |
|-------|----------|
| `.content` | Доступ к содержимому тега `<template>` |
| `.cloneNode()` | Создаёт копию узла |
| `true` | Глубокое клонирование (со всеми вложенными элементами) |

### 💡 **Пример без template (альтернатива):**
```javascript
// Вместо template можно создать элемент вручную:
const newLi = document.createElement('li');
newLi.className = 'todo-list-item';
newLi.innerHTML = `
    <label>
        <input type="checkbox" class="todo-list-input">
        <span>${taskText}</span>
    </label>
`;
```

### 🔍 **Свойство `textContent`**

```javascript
taskSpan.textContent = taskText;
```

| Свойство | Что делает | Пример |
|----------|------------|--------|
| `textContent` | Устанавливает/получает ТОЛЬКО текст | `span.textContent = 'Привет'` |
| `innerHTML` | Устанавливает/получает HTML + текст | `div.innerHTML = '<b>Жирный</b>'` |

⚠️ **Важно:** `textContent` безопаснее, так как не интерпретирует HTML-теги как код.

### 🔍 **Событие `change` для чекбокса**

```javascript
taskCheckbox.addEventListener('change', function() {
    if (this.checked) {
        taskItem.remove();
        updateEmptyMessage();
    }
});
```

| Часть | Значение |
|-------|----------|
| `addEventListener` | Назначает обработчик события |
| `'change'` | Событие — изменение состояния (чекбокса) |
| `this.checked` | `true` если чекбокс отмечен, `false` если нет |
| `.remove()` | Удаляет элемент из DOM полностью |

### 📊 **Жизненный цикл задачи:**
```
1. Пользователь вводит текст → 2. Создаётся копия template → 3. Заполняется текст
→ 4. Добавляется в список → 5. При клике на чекбокс → 6. Задача удаляется
```

## 4. **Обработчик формы**

```javascript
form.addEventListener('submit', function(event) {
    event.preventDefault();
    
    const taskText = input.value.trim();
    
    if (taskText !== '') {
        addTask(taskText);
        input.value = '';
        input.focus();
    }
});
```

### 🔍 **Событие `submit`**

Срабатывает, когда:
- Пользователь нажимает Enter в поле ввода
- Пользователь кликает на кнопку `type="submit"`

### 🔍 **`event.preventDefault()`**

```javascript
event.preventDefault();
```

**Что делает:** Отменяет стандартное поведение браузера:
- ❌ Без `preventDefault()`: страница перезагружается
- ✅ С `preventDefault()`: страница НЕ перезагружается, форма работает через JS

### 💡 **Пример без формы (альтернатива):**
```javascript
// Можно обрабатывать клик по кнопке, а не submit формы
button.addEventListener('click', function() {
    // та же логика
});
```

### 🔍 **Метод `trim()`**

```javascript
const taskText = input.value.trim();
```

| До `trim()` | После `trim()` |
|-------------|----------------|
| `"   Привет   "` | `"Привет"` |
| `"   "` (только пробелы) | `""` (пустая строка) |
| `"Купить хлеб"` | `"Купить хлеб"` |

**Зачем?** Убирает лишние пробелы в начале и конце, чтобы пользователь не мог добавить пустую задачу.

### 🔍 **Метод `focus()`**

```javascript
input.focus();
```

**Что делает:** Устанавливает курсор в поле ввода, чтобы пользователь мог сразу печатать новую задачу.

## 5. **Обработка существующих задач**

```javascript
document.querySelectorAll('.todo-list-item .todo-list-input').forEach(checkbox => {
    checkbox.addEventListener('change', function() {
        if (this.checked) {
            this.closest('.todo-list-item').remove();
            updateEmptyMessage();
        }
    });
});
```

### 🔍 **Метод `closest()`**

```javascript
this.closest('.todo-list-item')
```

**Что делает:** Ищет ближайший родительский элемент с указанным селектором.

### 📊 **Визуализация `closest`:**
```html
<li class="todo-list-item">          ← closest найдёт этот
  <label>
    <input class="todo-list-input">   ← this (чекбокс)
  </label>
</li>
```

### 💡 **Чем отличается от `parentElement`?**
```javascript
// parentElement — только прямой родитель
this.parentElement.parentElement; // <li>? зависит от структуры

// closest — ищет любого родителя с классом
this.closest('.todo-list-item'); // всегда найдёт <li>
```

## 6. **Полная схема работы приложения**

```
📄 Загрузка страницы
    ↓
🔍 Находим все элементы (form, input, список, сообщение, template)
    ↓
📝 Назначаем обработчики на существующие чекбоксы
    ↓
👁️ Проверяем, пуст ли список (updateEmptyMessage)
    ↓
⏸️ Ожидаем действий пользователя
    ↓
┌─────────────────────────────────────┐
│  Действия пользователя:              │
├─────────────────────────────────────┤
│ ✏️ Ввод текста + submit             │
│    ↓                                 │
│    🔍 Проверяем, что текст не пустой │
│    ↓                                 │
│    ➕ addTask() → добавляем задачу   │
│    ↓                                 │
│    🔄 Обновляем сообщение            │
├─────────────────────────────────────┤
│ ☑️ Клик на чекбокс                   │
│    ↓                                 │
│    ❌ Удаляем задачу                 │
│    ↓                                 │
│    🔄 Обновляем сообщение            │
└─────────────────────────────────────┘
```
