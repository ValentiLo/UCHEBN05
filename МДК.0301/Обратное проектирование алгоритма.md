**Обратное проектирование (reverse engineering) алгоритмов** в Python. Это процесс анализа готового кода или программы для понимания принципов его работы.

## **Что такое обратное проектирование?**

Это когда вы:
- Анализируете чужой код, чтобы понять алгоритм
- Восстанавливаете логику по результатам работы
- Изучаете, как работает "черный ящик"

```python
def mystery_function(data: list) -> list:
    result = []
    for i in range(len(data)):
        for j in range(i + 1, len(data)):
            if data[i] > data[j]:
                data[i], data[j] = data[j], data[i]
```

### **Анализ "черного ящика"**

```python
import random
from typing import List, Callable

class BlackBox:
    """Загадочный класс, который нужно понять"""
    def __init__(self, seed=None):
        if seed:
            random.seed(seed)
        self.secret = random.randint(1, 100)
    
    def process(self, numbers: List[int]) -> int:
        """
        Что делает этот метод?
        Нужно понять по входным и выходным данным
        """
        result = 0
        for num in numbers:
            if num > self.secret:
                result += 1
        return result

def reverse_engineer_blackbox():
    """
    Восстанавливаем логику BlackBox через эксперименты
    """
    print("🔍 Начинаем обратное проектирование BlackBox")
    print("=" * 50)
    
    # Эксперимент 1: пустой список
    box = BlackBox(seed=42)  # Фиксируем seed для повторяемости
    result = box.process([])
    print(f"Эксперимент 1: process([]) = {result}")
    print("  Вывод: при пустом списке всегда 0")
    
    # Эксперимент 2: все числа маленькие
    result = box.process([1, 2, 3, 4, 5])
    print(f"Эксперимент 2: process([1,2,3,4,5]) = {result}")
    
    # Эксперимент 3: все числа большие
    result = box.process([100, 200, 300])
    print(f"Эксперимент 3: process([100,200,300]) = {result}")
    
    # Эксперимент 4: смешанные числа
    result = box.process([50, 60, 70, 1, 2, 3])
    print(f"Эксперимент 4: process([50,60,70,1,2,3]) = {result}")
    
    # Эксперимент 5: определяем пороговое значение
    print("\n🎯 Ищем пороговое значение:")
    for test in range(1, 101):
        result = box.process([test])
        if result == 1:
            print(f"  Найдено! Порог = {test}")
            break
    
    print("\n✅ Вывод: BlackBox считает количество чисел, превышающих")
    print("   секретное пороговое значение, установленное при инициализации")

reverse_engineer_blackbox()
```

```python
import time
from typing import List, Tuple

def reverse_engineer_complex_algorithm():
    """
    Анализируем сложный алгоритм, определяем его сложность
    и восстанавливаем математическую формулу
    """
    
    def mysterious_algorithm(n: int) -> int:
        """Загадочный алгоритм"""
        if n <= 1:
            return 1
        
        result = 0
        for i in range(n):
            for j in range(n):
                result += i * j
        return result + mysterious_algorithm(n // 2)
    
    # Собираем данные о времени выполнения
    times = []
    results = []
    sizes = list(range(1, 100, 10))
    
    print("📊 Сбор данных о производительности:")
    print("Размер | Результат | Время (сек)")
    print("-" * 40)
    
    for n in sizes:
        start = time.time()
        result = mysterious_algorithm(n)
        elapsed = time.time() - start
        times.append(elapsed)
        results.append(result)
        print(f"{n:6} | {result:9} | {elapsed:.6f}")
    
    # Анализируем рост времени
    print("\n📈 Анализ сложности:")
    for i in range(1, len(times)):
        ratio = times[i] / times[i-1]
        size_ratio = sizes[i] / sizes[i-1]
        print(f"n={sizes[i-1]}→{sizes[i]}: время увеличилось в {ratio:.2f} раз")
    
    # Восстанавливаем формулу
    print("\n🎯 Восстановление математической формулы:")
    for i, result in enumerate(results[:5]):
        n = sizes[i]
        print(f"n={n}: результат={result}")
        # Проверяем гипотезы
        print(f"  n^3 = {n**3}")
        print(f"  n^2*log(n) = {n**2 * (n and n.bit_length())}")
    
    print("\n💡 Вывод: Алгоритм имеет сложность O(n² * log n)")
    print("   Формула: T(n) = O(n²) + T(n/2)")
```

## **6. Инструменты для обратного проектирования**

```python
import ast
import inspect
import sys
from typing import Any

class CodeAnalyzer:
    """Инструменты для анализа кода"""
    
    @staticmethod
    def get_call_graph(func, depth=0, max_depth=3):
        """Строит граф вызовов функции"""
        if depth > max_depth:
            return
        
        indent = "  " * depth
        print(f"{indent}→ {func.__name__}")
        
        # Получаем исходный код
        try:
            source = inspect.getsource(func)
            tree = ast.parse(source)
            
            # Находим вызовы других функций
            for node in ast.walk(tree):
                if isinstance(node, ast.Call):
                    if isinstance(node.func, ast.Name):
                        callee_name = node.func.id
                        if callee_name in globals():
                            callee = globals()[callee_name]
                            if callable(callee):
                                CodeAnalyzer.get_call_graph(
                                    callee, depth + 1, max_depth
                                )
        except:
            pass
    
    @staticmethod
    def analyze_data_flow(func, *args):
        """Анализирует поток данных через функцию"""
        print(f"Анализ потока данных в {func.__name__}")
        print("=" * 50)
        
        # Используем отладчик
        import pdb
        import sys
        from io import StringIO
        
        # Перехватываем вывод
        old_stdout = sys.stdout
        sys.stdout = StringIO()
        
        try:
            # Выполняем функцию с отладкой
            result = func(*args)
        finally:
            sys.stdout = old_stdout
        
        print(f"Входные данные: {args}")
        print(f"Выходные данные: {result}")
        
        return result
    
    @staticmethod
    def measure_complexity(func, *args):
        """Измеряет временную сложность"""
        import time
        
        def run_test(n):
            if args:
                test_args = (n,) + args[1:]
            else:
                test_args = (n,)
            
            start = time.time()
            result = func(*test_args)
            return time.time() - start, result
        
        print(f"Анализ сложности {func.__name__}")
        sizes = [10, 100, 1000, 10000]
        
        for size in sizes:
            try:
                elapsed, _ = run_test(size)
                print(f"  n={size:6}: {elapsed:.6f} сек")
            except RecursionError:
                print(f"  n={size}: превышена глубина рекурсии")
                break

# Пример использования
def fibonacci(n):
    if n <= 1:
        return n
    return fibonacci(n-1) + fibonacci(n-2)

analyzer = CodeAnalyzer()
print("Граф вызовов:")
analyzer.get_call_graph(fibonacci, max_depth=2)

print("\nПоток данных:")
analyzer.analyze_data_flow(fibonacci, 5)

print("\nАнализ сложности:")
analyzer.measure_complexity(fibonacci)
```

## **7. Практический пример: Reverse Engineering API**

```python
import requests
import json
from typing import Dict, Any

class APIReverser:
    """Инструмент для обратного проектирования API"""
    
    def __init__(self, base_url: str):
        self.base_url = base_url
        self.endpoints = {}
        self.observations = []
    
    def probe_endpoint(self, endpoint: str, method: str = 'GET', 
                       data: Dict = None) -> Dict[str, Any]:
        """Исследует эндпоинт"""
        url = f"{self.base_url}{endpoint}"
        
        print(f"\n🔍 Исследуем {method} {url}")
        
        try:
            if method == 'GET':
                response = requests.get(url, params=data, timeout=5)
            elif method == 'POST':
                response = requests.post(url, json=data, timeout=5)
            else:
                return {'error': 'Unsupported method'}
            
            # Анализируем ответ
            result = {
                'status': response.status_code,
                'headers': dict(response.headers),
                'content_type': response.headers.get('content-type', ''),
                'data': None
            }
            
            # Пытаемся распарсить ответ
            if 'application/json' in result['content_type']:
                result['data'] = response.json()
            else:
                result['data'] = response.text[:200]  # Первые 200 символов
            
            # Запоминаем наблюдения
            self.observations.append({
                'endpoint': endpoint,
                'method': method,
                'data': data,
                'response': result
            })
            
            return result
            
        except Exception as e:
            return {'error': str(e)}
    
    def analyze_patterns(self):
        """Анализирует паттерны в найденных эндпоинтах"""
        print("\n📊 Анализ паттернов:")
        
        # Группируем по статусам
        statuses = {}
        for obs in self.observations:
            status = obs['response'].get('status')
            statuses[status] = statuses.get(status, 0) + 1
        
        print(f"Статус-коды: {statuses}")
        
        # Анализируем структуру данных
        for obs in self.observations:
            if obs['response'].get('data'):
                data = obs['response']['data']
                if isinstance(data, dict):
                    print(f"\n{obs['endpoint']}:")
                    print(f"  Ключи: {list(data.keys())}")
                    for key, value in data.items():
                        print(f"  {key}: {type(value).__name__}")
    
    def guess_schema(self, endpoint: str) -> Dict:
        """Пытается угадать схему данных"""
        print(f"\n🎯 Угадываем схему для {endpoint}")
        
        # Пробуем разные варианты
        tests = [
            {'id': 1},
            {'page': 1, 'limit': 10},
            {'search': 'test'},
            {'sort': 'asc'}
        ]
        
        schemas = []
        for test in tests:
            result = self.probe_endpoint(endpoint, 'GET', test)
            if result.get('status') == 200:
                schemas.append(test)
        
        return {
            'possible_params': schemas,
            'endpoint': endpoint
        }

# Пример использования (с публичным API)
def reverse_engineer_public_api():
    """Обратное проектирование публичного API"""
    
    reverser = APIReverser("https://jsonplaceholder.typicode.com")
    
    # Исследуем эндпоинты
    print("🔍 Начинаем разведку API")
    print("=" * 50)
    
    # Пробуем стандартные эндпоинты
    endpoints = ['/posts', '/users', '/comments', '/posts/1']
    
    for endpoint in endpoints:
        result = reverser.probe_endpoint(endpoint)
        if result.get('status') == 200:
            print(f"✅ {endpoint} доступен")
            if result.get('data') and isinstance(result['data'], list):
                print(f"   Найдено {len(result['data'])} записей")
                if result['data']:
                    first_item = result['data'][0]
                    print(f"   Структура: {list(first_item.keys())}")
        else:
            print(f"❌ {endpoint} недоступен: {result.get('status')}")
    
    # Анализируем паттерны
    reverser.analyze_patterns()
    
    # Угадываем схему
    schema = reverser.guess_schema('/posts')
    print(f"\n💡 Предполагаемая схема: {schema}")

# Запускаем (закомментировано, так как требует интернет)
```
