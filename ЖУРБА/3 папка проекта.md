# **Что должно быть в папке проекта для запуска**

Вот **полная структура папки проекта**, которая должна быть у тебя:

```
db-analyzer/                    ← Главная папка проекта
│
├── 📁 backend/                 ← Серверная часть (Python)
│   ├── 📄 main.py             ← Главный файл сервера
│   ├── 📄 requirements.txt    ← Список библиотек Python
│   ├── 📁 src/               ← Исходный код
│   │   ├── 📁 analyzers/     ← Анализаторы SQL
│   │   ├── 📁 models/        ← Модели данных
│   │   └── 📁 utils/         ← Вспомогательные функции
│   └── 📁 tests/             ← Тесты
│
├── 📁 frontend/               ← Веб-интерфейс (React)
│   ├── 📄 package.json       ← Настройки и зависимости
│   ├── 📄 package-lock.json  ← Фиксация версий
│   ├── 📁 public/           ← Статические файлы
│   │   └── 📄 index.html    ← Главная HTML страница
│   └── 📁 src/              ← Исходный код React
│       ├── 📄 App.js        ← Главный компонент
│       ├── 📄 index.js      ← Точка входа
│       └── 📁 components/   ← Компоненты интерфейса
│
├── 📁 docker/                 ← Конфигурация Docker
│   ├── 📄 Dockerfile.backend ← Образ для бэкенда
│   └── 📄 Dockerfile.frontend ← Образ для фронтенда
│
├── 📄 docker-compose.yml      ← Одна команда для запуска всего
├── 📄 .env.example           ← Пример настроек (переименовать в .env)
├── 📄 README.md              ← Инструкция по установке
├── 📄 .gitignore             ← Что не отправлять в Git
└── 📄 LICENSE               ← Лицензия проекта
```

## **2. Что делать, если файлов нет?**

### **Вариант 1: Создать минимальный проект самому**

**Шаг 1: Создай структуру папок:**
```bash
mkdir db-analyzer
cd db-analyzer
mkdir backend frontend
```

**Шаг 2: Создай docker-compose.yml:**
```yaml
# Скопируй этот текст в файл docker-compose.yml
version: '3.8'

services:
  backend:
    image: python:3.11-slim
    command: sh -c "pip install fastapi uvicorn && uvicorn main:app --host 0.0.0.0 --port 8000"
    ports:
      - "8000:8000"
    volumes:
      - ./backend:/app
    working_dir: /app

  frontend:
    image: node:18-alpine
    command: sh -c "npm install && npm start"
    ports:
      - "3000:3000"
    volumes:
      - ./frontend:/app
    working_dir: /app
    environment:
      - REACT_APP_API_URL=http://backend:8000
```

**Шаг 3: Создай backend/main.py:**
```python
from fastapi import FastAPI
from fastapi.middleware.cors import CORSMiddleware

app = FastAPI()

# Разрешить запросы от фронтенда
app.add_middleware(
    CORSMiddleware,
    allow_origins=["http://localhost:3000"],
    allow_credentials=True,
    allow_methods=["*"],
    allow_headers=["*"],
)

@app.get("/")
def read_root():
    return {"message": "DB Analyzer Backend is running!"}

@app.get("/health")
def health_check():
    return {"status": "healthy"}

@app.post("/analyze")
def analyze_sql(sql: str):
    return {
        "sql": sql,
        "analysis": "Это анализ SQL запроса",
        "suggestions": ["Используйте конкретные колонки вместо SELECT *"]
    }
```

**Шаг 4: Создай frontend/src/App.js:**
```jsx
import React, { useState } from 'react';
import axios from 'axios';

function App() {
  const [sql, setSql] = useState('');
  const [result, setResult] = useState(null);

  const analyzeSQL = async () => {
    try {
      const response = await axios.post('http://localhost:8000/analyze', {
        sql: sql
      });
      setResult(response.data);
    } catch (error) {
      setResult({ error: error.message });
    }
  };

  return (
    <div style={{ padding: '20px' }}>
      <h1>DB Analyzer</h1>
      
      <textarea
        value={sql}
        onChange={(e) => setSql(e.target.value)}
        placeholder="Введите SQL запрос..."
        rows="5"
        cols="50"
      />
      
      <br />
      <button onClick={analyzeSQL}>Анализировать</button>
      
      {result && (
        <div style={{ marginTop: '20px' }}>
          <h3>Результат:</h3>
          <pre>{JSON.stringify(result, null, 2)}</pre>
        </div>
      )}
    </div>
  );
}

export default App;
```
